    67→    67→function lineTotalCents(quantity: string, unitPrice: string, discountPercent: string): number {
    68→    68→  const qty = parseFloat(quantity) || 0;
    69→    69→  const cents = Math.round((parseFloat(unitPrice) || 0) * 100);
    70→    70→  const disc = parseFloat(discountPercent) || 0;
    71→    71→  return Math.round(qty * cents * (1 - disc / 100));
    72→    72→}
    73→    73→
    74→    74→function formTotals(items: FormItem[], discount: string, shipping: string) {
    75→    75→  const subtotalCents = items.reduce((s, it) => s + lineTotalCents(it.quantity, it.unitPrice, it.discountPercent), 0);
    76→    76→  const taxCents = items.reduce((s, it) => s + Math.round((lineTotalCents(it.quantity, it.unitPrice, it.discountPercent) * (parseFloat(it.taxPercent) || 0)) / 100), 0);
    77→    77→  const discountCents = Math.round((parseFloat(discount) || 0) * 100);
    78→    78→  const shippingCents = Math.round((parseFloat(shipping) || 0) * 100);
    79→    79→  const totalCents = Math.max(0, subtotalCents - discountCents + taxCents + shippingCents);
    80→    80→  return { subtotalCents, taxCents, discountCents, shippingCents, totalCents };
    81→    81→}

    83→    83→/** Resolve the company label from /api/v1/settings with a graceful fallback. */
    84→    84→async function loadCompanyName(): Promise<string> {
    85→    85→  try {
    86→    86→    const res = await api.get<unknown>("/api/v1/settings");
    87→    87→    const d = res.data;
    88→    88→    if (Array.isArray(d)) {
    89→    89→      const row = d.find((r) => (r as { key?: string })?.key === "company_name") as { value?: string } | undefined;
    90→    90→      if (row?.value) return row.value;
    91→    91→    } else if (d && typeof d === "object") {
    92→    92→      const obj = d as Record<string, unknown>;
    93→    93→      if (typeof obj.company_name === "string") return obj.company_name;
    94→    94→      if (obj.company && typeof obj.company === "object" && typeof (obj.company as Record<string, unknown>).name === "string") {
    95→    95→        return String((obj.company as Record<string, unknown>).name);
    96→    96→      }
    97→    97→    }
    98→    98→  } catch {
    99→    99→    /* settings API not available yet — fallback label */
   100→   100→  }
   101→   101→  return FALLBACK_COMPANY;
   102→   102→}
