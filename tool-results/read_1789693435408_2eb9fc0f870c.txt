   117→function DraftBanner({
   118→  draftExists,
   119→  onRestore,
   120→  onDiscard,
   121→}: {
   122→  draftExists: boolean;
   123→  onRestore: () => void;
   124→  onDiscard: () => void;
   125→}) {
   126→  if (!draftExists) return null;
   127→  return (
   128→    <div className="flex flex-wrap items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-800">
   129→      <span className="font-medium">Unsaved draft found.</span>
   130→      <Button type="button" size="sm" variant="outline" className="h-7" onClick={onRestore}>
   131→        Restore
   132→      </Button>
   133→      <Button type="button" size="sm" variant="ghost" className="h-7 text-amber-800" onClick={onDiscard}>
   134→        Discard
   135→      </Button>
   136→    </div>
   137→  );
   138→}

   140→function num(value: string, fallback = 0): number {
   141→  const n = parseFloat(value);
   142→  return isFinite(n) ? n : fallback;
   143→}

   150→  category: string;

   152→  stockQty: string;
   153→  minStockQty: string;

   159→  sku: "",
   160→  name: "",
   161→  category: "",

   164→  minStockQty: "",

   169→export function InventoryModule() {
   170→  const { user } = useSession();
   171→  const { toast } = useToast();
   172→  const canManage = hasPerm(user, PERMISSIONS.inventory_manage satisfies Permission);
   173→  const canSupplier = hasPerm(user, PERMISSIONS.purchases_manage satisfies Permission);
   174→
   175→  const [tab, setTab] = useState("items");
   176→
   177→  // Items
   178→  const [items, setItems] = useState<ItemRow[] | null>(null);
   179→  const [stats, setStats] = useState<InventoryStats | null>(null);
   180→  const [itemsErr, setItemsErr] = useState("");
   181→  const [itemsLoading, setItemsLoading] = useState(true);
   182→
   183→  // Movements
   184→  const [movements, setMovements] = useState<MovementRow[] | null>(null);
   185→  const [movErr, setMovErr] = useState("");
   186→  const [movLoading, setMovLoading] = useState(false);
   187→
   188→  // Suppliers
   189→  const [suppliers, setSuppliers] = useState<SupplierRow[] | null>(null);
   190→  const [supErr, setSupErr] = useState("");
   191→  const [supLoading, setSupLoading] = useState(false);
   192→
   193→  const loadItems = useCallback(async () => {
   194→    setItemsLoading(true);
   195→    setItemsErr("");
   196→    try {
   197→      const res = await api.get<ItemRow[]>(`/api/v1/inventory${qs({ pageSize: 200 })}`);
   198→      setItems(res.data);
   199→      const s = (res.meta as { stats?: InventoryStats } | undefined)?.stats;
   200→      if (s) setStats(s);
   201→    } catch (e) {
   202→      setItemsErr(errMessage(e));
   203→    } finally {
   204→      setItemsLoading(false);
   205→    }
   206→  }, []);
   207→
   208→  const loadMovements = useCallback(async () => {
   209→    setMovLoading(true);
   210→    setMovErr("");
   211→    try {
   212→      const res = await api.get<MovementRow[]>(`/api/v1/inventory/movements${qs({ pageSize: 200 })}`);
   213→      setMovements(res.data);
   214→    } catch (e) {
   215→      setMovErr(errMessage(e));
   216→    } finally {
   217→      setMovLoading(false);
   218→    }
   219→  }, []);
   220→
   221→  const loadSuppliers = useCallback(async () => {
   222→    setSupLoading(true);
   223→    setSupErr("");
   224→    try {
   225→      const res = await api.get<SupplierRow[]>(`/api/v1/suppliers${qs({ pageSize: 200 })}`);
   226→      setSuppliers(res.data);
   227→    } catch (e) {
   228→      setSupErr(errMessage(e));
   229→    } finally {
   230→      setSupLoading(false);
   231→    }
   232→  }, []);
   233→
   234→  useEffect(() => {
   235→    loadItems();
   236→    loadSuppliers(); // needed for item create/edit supplier pickers
   237→  }, [loadItems, loadSuppliers]);
   238→
   239→  function onTabChange(value: string) {
   240→    setTab(value);
   241→    if (value === "movements" && !movements && !movLoading) loadMovements();
   242→    if (value === "suppliers" && !suppliers && !supLoading) loadSuppliers();
   243→  }
   244→
   245→  // ── Add item dialog (draft-protected) ──
   246→  const [addOpen, setAddOpen] = useState(false);
   247→  const [saving, setSaving] = useState(false);
   248→  const draft = useDraft<ItemDraft>({ formKey: "inventory.item.create", initial: BLANK_ITEM_DRAFT });
   249→
   250→  async function submitItem() {
   251→    const v = draft.value;
   252→    if (!v.name.trim()) {
   253→      toast({ title: "Name is required", variant: "destructive" });
   254→      return;
   255→    }
   256→    setSaving(true);
   257→    try {
   258→      await api.post<ItemRow>("/api/v1/inventory", {
   259→        sku: v.sku.trim() || undefined,
   260→        name: v.name.trim(),
   261→        category: v.category.trim() || undefined,
   262→        unit: v.unit.trim() || undefined,
   263→        stockQty: num(v.stockQty),
   264→        minStockQty: num(v.minStockQty),
   265→        unitCost: toCents(v.unitCost || "0") / 100,
   266→        supplierId: v.supplierId || undefined,
   267→      });
   268→      toast({ title: "Item created", description: `${v.name.trim()} added to inventory.` });
   269→      draft.reset(BLANK_ITEM_DRAFT);
   270→      setAddOpen(false);
   271→      loadItems();
   272→    } catch (e) {
   273→      toast({ title: "Could not create item", description: errMessage(e), variant: "destructive" });
   274→    } finally {
   275→      setSaving(false);
   276→    }
   277→  }
   278→
   279→  // ── Edit item dialog ──
   280→  const [editItem, setEditItem] = useState<ItemRow | null>(null);
   281→  const [editForm, setEditForm] = useState({ name: "", category: "", unit: "", minStockQty: "", unitCost: "", supplierId: "", status: "ACTIVE" });
   282→  const [editSaving, setEditSaving] = useState(false);
   283→
   284→  function openEdit(item: ItemRow) {
   285→    setEditForm({
   286→      name: item.name,
   287→      category: item.category,
   288→      unit: item.unit,
   289→      minStockQty: String(item.minStockQty),
   290→      unitCost: fromCents(item.unitCostCents),
   291→      supplierId: item.supplierId ?? "NONE",
   292→      status: item.status,
   293→    });
   294→    setEditItem(item);
   295→  }
   296→
   297→  async function submitEdit() {
   298→    if (!editItem) return;
   299→    setEditSaving(true);
   300→    try {
   301→      await api.patch<ItemRow>(`/api/v1/inventory/${editItem.id}`, {
   302→        name: editForm.name.trim(),
   303→        category: editForm.category.trim(),
   304→        unit: editForm.unit.trim(),
   305→        minStockQty: num(editForm.minStockQty),
   306→        unitCost: num(editForm.unitCost),
   307→        supplierId: editForm.supplierId === "NONE" ? null : editForm.supplierId,
   308→        status: editForm.status,
   309→      });
   310→      toast({ title: "Item updated", description: editItem.sku });
   311→      setEditItem(null);
   312→      loadItems();
   313→    } catch (e) {
   314→      toast({ title: "Could not update item", description: errMessage(e), variant: "destructive" });
   315→    } finally {
   316→      setEditSaving(false);
   317→    }
   318→  }
   319→
   320→  // ── Adjust stock dialog ──
   321→  const [adjustItem, setAdjustItem] = useState<ItemRow | null>(null);
   322→  const [adjustType, setAdjustType] = useState<(typeof MOVEMENT_TYPES)[number]>("RECEIVE");
   323→  const [adjustQty, setAdjustQty] = useState("");
   324→  const [adjustNote, setAdjustNote] = useState("");
   325→  const [adjustSaving, setAdjustSaving] = useState(false);
   326→
   327→  function openAdjust(item: ItemRow) {
   328→    setAdjustType("RECEIVE");
   329→    setAdjustQty("");
   330→    setAdjustNote("");
   331→    setAdjustItem(item);
   332→  }
   333→
   334→  async function submitAdjust() {
   335→    if (!adjustItem) return;
   336→    const qty = num(adjustQty, NaN);
   337→    if (!isFinite(qty) || qty === 0 || (adjustType !== "ADJUST" && qty <= 0)) {
   338→      toast({
   339→        title: "Invalid quantity",
   340→        description: adjustType === "ADJUST" ? "Enter a non-zero signed delta (e.g. -2 or 5)." : "Enter a positive quantity.",
   341→        variant: "destructive",
   342→      });
   343→      return;
   344→    }
   345→    setAdjustSaving(true);
   346→    try {
   347→      const res = await api.post<{ item: ItemRow }>(`/api/v1/inventory/${adjustItem.id}/movement`, {
   348→        type: adjustType,
   349→        quantity: qty,
   350→        note: adjustNote.trim() || undefined,
   351→      });
   352→      toast({
   353→        title: "Stock updated",
   354→        description: `${adjustItem.sku} — new balance ${res.data.item.stockQty} ${adjustItem.unit}.`,
   355→      });
   356→      setAdjustItem(null);
   357→      loadItems();
   358→      if (movements) loadMovements();
   359→    } catch (e) {
   360→      toast({ title: "Could not record movement", description: errMessage(e), variant: "destructive" });
   361→    } finally {
   362→      setAdjustSaving(false);
   363→    }
   364→  }
   365→
   366→  // ── Delete item ──
   367→  const [deleteItem, setDeleteItem] = useState<ItemRow | null>(null);
   368→  const [deleteSaving, setDeleteSaving] = useState(false);
   369→
   370→  async function submitDelete() {
   371→    if (!deleteItem) return;
   372→    setDeleteSaving(true);
   373→    try {
   374→      await api.del(`/api/v1/inventory/${deleteItem.id}`);
   375→      toast({ title: "Item removed", description: `${deleteItem.sku} archived or deleted.` });
   376→      setDeleteItem(null);
   377→      loadItems();
   378→    } catch (e) {
   379→      toast({ title: "Could not remove item", description: errMessage(e), variant: "destructive" });
   380→    } finally {
   381→      setDeleteSaving(false);
   382→    }
   383→  }
   384→
   385→  // ── Supplier create/edit dialog ──
   386→  const [supDialog, setSupDialog] = useState<{ open: boolean; supplier: SupplierRow | null }>({ open: false, supplier: null });
   387→  const [supForm, setSupForm] = useState({ name: "", contactPerson: "", email: "", phone: "", address: "" });
   388→  const [supSaving, setSupSaving] = useState(false);
   389→  const [deleteSupplier, setDeleteSupplier] = useState<SupplierRow | null>(null);
   390→
   391→  function openSupplierDialog(supplier: SupplierRow | null) {
   392→    setSupForm(
   393→      supplier
   394→        ? { name: supplier.name, contactPerson: supplier.contactPerson, email: supplier.email, phone: supplier.phone, address: supplier.address }
   395→        : { name: "", contactPerson: "", email: "", phone: "", address: "" }
   396→    );
   397→    setSupDialog({ open: true, supplier });
   398→  }
   399→
   400→  async function submitSupplier() {
   401→    if (!supForm.name.trim()) {
   402→      toast({ title: "Name is required", variant: "destructive" });
   403→      return;
   404→    }
   405→    setSupSaving(true);
   406→    try {
   407→      if (supDialog.supplier) {
   408→        await api.patch(`/api/v1/suppliers/${supDialog.supplier.id}`, supForm);
   409→        toast({ title: "Supplier updated" });
   410→      } else {
   411→        await api.post("/api/v1/suppliers", supForm);
   412→        toast({ title: "Supplier created" });
   413→      }
   414→      setSupDialog({ open: false, supplier: null });
   415→      loadSuppliers();
   416→    } catch (e) {
   417→      toast({ title: "Could not save supplier", description: errMessage(e), variant: "destructive" });
   418→    } finally {
   419→      setSupSaving(false);
   420→    }
   421→  }
   422→
   423→  async function submitDeleteSupplier() {
   424→    if (!deleteSupplier) return;
   425→    try {
   426→      await api.del(`/api/v1/suppliers/${deleteSupplier.id}`);
   427→      toast({ title: "Supplier removed" });
   428→      setDeleteSupplier(null);
   429→      loadSuppliers();
   430→      loadItems();
   431→    } catch (e) {
   432→      toast({ title: "Could not remove supplier", description: errMessage(e), variant: "destructive" });
   433→    }
   434→  }
   435→
   436→  // ── table columns ──
   437→
   438→  const itemColumns: Column<ItemRow>[] = [
   439→    { key: "sku", header: "SKU", value: (r) => r.sku, render: (r) => <span className="font-mono text-xs">{r.sku}</span> },
   440→    { key: "name", header: "Item", value: (r) => r.name, render: (r) => <span className="font-medium">{r.name}</span> },
   441→    { key: "category", header: "Category", value: (r) => r.category, hideOnMobile: true },
   442→    {
   443→      key: "stockQty",
   444→      header: "Stock",
   445→      value: (r) => r.stockQty,
   446→      render: (r) => (
   447→        <span className={cn("tabular-nums font-medium", r.low && "text-red-600")}>
   448→          {r.stockQty} {r.unit}
   449→          {r.low ? " ⚠" : ""}
   450→        </span>
   451→      ),
   452→    },
   453→    { key: "minStockQty", header: "Min", value: (r) => r.minStockQty, className: "tabular-nums", hideOnMobile: true },
   454→    { key: "unitCostCents", header: "Unit cost", value: (r) => r.unitCostCents / 100, render: (r) => money(r.unitCostCents), className: "tabular-nums", hideOnMobile: true },
   455→    { key: "supplier", header: "Supplier", value: (r) => r.supplier?.name ?? "", render: (r) => r.supplier?.name ?? "—", hideOnMobile: true },
   456→    { key: "status", header: "Status", value: (r) => r.status, render: (r) => <StatusBadge status={r.status} /> },
   457→    ...(canManage
   458→      ? [
   459→          {
   460→            key: "actions",
   461→            header: "",
   462→            sortable: false,
   463→            className: "w-[110px] text-right",
   464→            render: (r: ItemRow) => (
   465→              <div className="flex justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   466→                <Button size="icon" variant="ghost" className="h-8 w-8" aria-label={`Adjust stock for ${r.sku}`} onClick={() => openAdjust(r)}>
   467→                  <PackagePlus className="h-4 w-4" />
   468→                </Button>
   469→                <Button size="icon" variant="ghost" className="h-8 w-8" aria-label={`Edit ${r.sku}`} onClick={() => openEdit(r)}>
   470→                  <Pencil className="h-4 w-4" />
   471→                </Button>
   472→                <Button size="icon" variant="ghost" className="h-8 w-8 text-red-600 hover:text-red-700" aria-label={`Remove ${r.sku}`} onClick={() => setDeleteItem(r)}>
   473→                  <Trash2 className="h-4 w-4" />
   474→                </Button>
   475→              </div>
   476→            ),
   477→          } satisfies Column<ItemRow>,
   478→        ]
   479→      : []),
   480→  ];
   481→
   482→  const movementColumns: Column<MovementRow>[] = [
   483→    { key: "createdAt", header: "Date", value: (r) => r.createdAt, render: (r) => fmtDateTime(r.createdAt), hideOnMobile: true },
   484→    {
   485→      key: "item",
   486→      header: "Item",
   487→      value: (r) => (r.item ? `${r.item.sku} ${r.item.name}` : r.itemId),
   488→      render: (r) => (
   489→        <div className="min-w-0">
   490→          <div className="font-medium truncate">{r.item?.name ?? "—"}</div>
   491→          <div className="font-mono text-xs text-muted-foreground">{r.item?.sku}</div>
   492→        </div>
   493→      ),
   494→    },
   495→    {
   496→      key: "type",
   497→      header: "Type",
   498→      value: (r) => r.type,
   499→      render: (r) => <Badge variant="outline" className={cn("border-transparent", MOVEMENT_TONE[r.type])}>{humanize(r.type)}</Badge>,
   500→    },
   501→    {
   502→      key: "quantity",
   503→      header: "Qty",
   504→      value: (r) => r.quantity,
   505→      render: (r) => (
   506→        <span className={cn("tabular-nums font-medium", r.quantity >= 0 ? "text-emerald-600" : "text-red-600")}>
   507→          {r.quantity >= 0 ? "+" : ""}
   508→          {r.quantity}
   509→        </span>
   510→      ),
   511→    },
   512→    { key: "balanceAfter", header: "Balance", value: (r) => r.balanceAfter, className: "tabular-nums" },
   513→    {
   514→      key: "note",
   515→      header: "Note",
   516→      value: (r) => r.note,
   517→      render: (r) => <span className="text-muted-foreground line-clamp-1 max-w-[220px]">{r.note || r.referenceType.toLowerCase()}</span>,
   518→      hideOnMobile: true,
   519→    },
   520→  ];
   521→
   522→  const supplierColumns: Column<SupplierRow>[] = [
   523→    { key: "code", header: "Code", value: (r) => r.code, render: (r) => <span className="font-mono text-xs">{r.code}</span> },
   524→    { key: "name", header: "Supplier", value: (r) => r.name, render: (r) => <span className="font-medium">{r.name}</span> },
   525→    { key: "contactPerson", header: "Contact", value: (r) => r.contactPerson, render: (r) => r.contactPerson || "—", hideOnMobile: true },
