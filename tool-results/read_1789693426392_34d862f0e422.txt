    67→    67→type FieldErrors = Record<string, string>;
    68→    68→function extractFieldErrors(e: unknown): FieldErrors {
    69→    69→  if (e instanceof ClientApiError && Array.isArray(e.details)) {
    70→    70→    const out: FieldErrors = {};
    71→    71→    for (const d of e.details as { path?: string; message?: string }[]) {
    72→    72→      if (d?.path && d?.message) out[d.path] = d.message;
    73→    73→    }
    74→    74→    return out;
    75→    75→  }
    76→    76→  return {};
    77→    77→}
    78→    78→
    79→    79→function FieldError({ msg }: { msg?: string }) {
    80→    80→  if (!msg) return null;
    81→    81→  return <p className="text-xs text-destructive mt-1">{msg}</p>;
    82→    82→}
