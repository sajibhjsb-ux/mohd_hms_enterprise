     1→"use client";
     2→
     3→// MOHD.HMS ENTERPRISE — dedicated Complaint Entry page (complaints/new view).
     4→// Replaces the old "New Complaint" modal as the primary complaint entry mechanism.
     5→// Reuses the existing Complaint API, RBAC, workflow, numbering, audit and draft
     6→// architecture — no duplicate tables, no duplicate APIs, no mock data.
     7→
     8→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     9→import { api, qs } from "@/lib/hms/api-client";
    10→import { hasPerm, useSession } from "@/components/hms/session";
    11→import { useUi } from "@/lib/hms/ui-store";
    12→import { navigateTo } from "@/lib/hms/router";
    13→import { useToast } from "@/hooks/use-toast";
    14→import { useDraft } from "@/hooks/use-draft";
    15→import {
    16→  PageHeader, EmptyState,
    17→} from "@/components/hms/shared/ui-bits";
    18→import { PERMISSIONS, PRIORITIES, humanize } from "@/lib/hms/constants";
    19→import { fmtDateTime } from "@/lib/hms/format";
    20→import { Button } from "@/components/ui/button";
    21→import { Input } from "@/components/ui/input";
    22→import { Label } from "@/components/ui/label";
    23→import { Textarea } from "@/components/ui/textarea";
    24→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    25→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    26→import {
    27→  AlertCircle, ArrowLeft, ChevronRight, Loader2, MapPin, Save, Send, User, X,
    28→} from "lucide-react";
    29→import { cn } from "@/lib/utils";
    30→
    31→// ── Types ──
    32→
    33→type CustomerOpt = {
    34→  id: string;
    35→  companyName?: string;
    36→  name?: string;
    37→  contactPerson?: string;
    38→  phone?: string;
    39→  code?: string;
    40→};
    41→
    42→type EquipmentOpt = {
    43→  id: string;
    44→  name?: string;
    45→  assetTag?: string;
    46→  location?: { id: string; name?: string | null; code?: string | null } | null;
    47→  customer?: { id: string; companyName?: string; code?: string } | null;
    48→};
    49→
    50→type CreateForm = {
    51→  title: string;
    52→  description: string;
    53→  priority: string;
    54→  customerId: string;
    55→  equipmentId: string;
    56→};
    57→
    58→const EMPTY_CREATE: CreateForm = { title: "", description: "", priority: "MEDIUM", customerId: "", equipmentId: "" };
    59→const TITLE_MAX = 200;
    60→const DESC_MAX = 5000;
    61→
    62→// ── Page ──
    63→
    64→export function ComplaintNewPage() {
    65→  const { user } = useSession();
    66→  const { toast } = useToast();
    67→  const setPageDirty = useUi((s) => s.setPageDirty);
    68→  const isStaffUser = !!user && user.role !== "CUSTOMER";
    69→  const canCreate = hasPerm(user, PERMISSIONS.complaints_create);
    70→
    71→  const draft = useDraft<CreateForm>({ formKey: "complaint.create", initial: EMPTY_CREATE });
    72→  const [errors, setErrors] = useState<Record<string, string>>({});
    73→  const [submitError, setSubmitError] = useState<string | null>(null);
    74→  const [creating, setCreating] = useState(false);
    75→
    76→  // ── Customer search (API-backed, never loads the whole table) ──
    77→  const [custQuery, setCustQuery] = useState("");
    78→  const [custResults, setCustResults] = useState<CustomerOpt[]>([]);
    79→  const [custLoading, setCustLoading] = useState(false);
    80→  const [custOpen, setCustOpen] = useState(false);
    81→  const [custInitialDone, setCustInitialDone] = useState(false);
    82→  const [selectedCustomer, setSelectedCustomer] = useState<CustomerOpt | null>(null);
    83→  const [portalCustomerName, setPortalCustomerName] = useState<string | null>(null);
    84→  const searchSeq = useRef(0);
    85→
    86→  const selectedId = isStaffUser ? draft.value.customerId : (user?.customerId ?? "");
    87→
    88→  // Initial page of customers (20) for staff — search narrows server-side from there.
    89→  useEffect(() => {
    90→    if (!isStaffUser) return;
    91→    let alive = true;
    92→    setCustLoading(true);
    93→    api.get<CustomerOpt[]>(`/api/v1/customers${qs({ pageSize: 20 })}`)
    94→      .then((r) => { if (alive) { setCustResults(Array.isArray(r.data) ? r.data : []); setCustInitialDone(true); } })
    95→      .catch(() => { if (alive) setCustInitialDone(true); })
    96→      .finally(() => { if (alive) setCustLoading(false); });
    97→    return () => { alive = false; };
    98→  }, [isStaffUser]);
    99→
   100→  // Debounced server-side search while typing (≥2 chars); empty query restores the initial page.
   101→  useEffect(() => {
   102→    if (!isStaffUser) return;
   103→    const q = custQuery.trim();
   104→    if (q.length < 2) {
   105→      if (custInitialDone && custQuery === "") {
   106→        let alive = true;
   107→        setCustLoading(true);
   108→        api.get<CustomerOpt[]>(`/api/v1/customers${qs({ pageSize: 20 })}`)
   109→          .then((r) => { if (alive) setCustResults(Array.isArray(r.data) ? r.data : []); })
   110→          .catch(() => undefined)
   111→          .finally(() => { if (alive) setCustLoading(false); });
   112→        return () => { alive = false; };
   113→      }
   114→      return;
   115→    }
   116→    const seq = ++searchSeq.current;
   117→    const t = setTimeout(() => {
   118→      setCustLoading(true);
   119→      api.get<CustomerOpt[]>(`/api/v1/customers${qs({ search: q, pageSize: 20 })}`)
   120→        .then((r) => { if (alive(seq)) setCustResults(Array.isArray(r.data) ? r.data : []); })
   121→        .catch(() => { if (alive(seq)) setCustResults([]); })
   122→        .finally(() => { if (alive(seq)) setCustLoading(false); });
   123→    }, 300);
   124→    function alive(s: number) { return s === searchSeq.current; }
   125→    return () => clearTimeout(t);
   126→  }, [custQuery, custInitialDone, isStaffUser]);
   127→
   128→  // Portal users file under their own customer record — show its name for clarity.
   129→  useEffect(() => {
   130→    if (isStaffUser || !user?.customerId) return;
   131→    let alive = true;
   132→    api.get<CustomerOpt[]>("/api/v1/customers")
   133→      .then((r) => {
   134→        if (!alive) return;
   135→        const own = Array.isArray(r.data) ? r.data[0] : undefined;
   136→        if (own) setPortalCustomerName(own.companyName ?? own.name ?? null);
   137→      })
   138→      .catch(() => undefined);
   139→    return () => { alive = false; };
   140→  }, [isStaffUser, user?.customerId]);
   141→
   142→  // BUGFIX: after a draft restores customerId (component-local display state is
   143→  // lost on reload), re-fetch the customer record so the selection card renders.
   144→  useEffect(() => {
   145→    if (!isStaffUser) return;
   146→    const cid = draft.value.customerId;
   147→    if (!cid || selectedCustomer?.id === cid) return;
   148→    let alive = true;
   149→    api.get<CustomerOpt[]>(`/api/v1/customers${qs({ customerId: cid })}`)
   150→      .then((r) => {
   151→        if (!alive) return;
   152→        const c = Array.isArray(r.data) ? r.data.find((x) => x.id === cid) : undefined;
   153→        if (c) {
   154→          setSelectedCustomer(c);
   155→          setCustQuery(c.companyName ?? c.name ?? c.id);
   156→        }
   157→      })
   158→      .catch(() => undefined);
   159→    return () => { alive = false; };
   160→  }, [draft.value.customerId, isStaffUser, selectedCustomer?.id]);
   161→
   162→  // ── Equipment follows the selected customer (server-filtered) ──
   163→  const [equipment, setEquipment] = useState<EquipmentOpt[]>([]);
   164→  const [equipLoading, setEquipLoading] = useState(false);
   165→  const [equipLoaded, setEquipLoaded] = useState(false);
   166→
   167→  useEffect(() => {
   168→    if (!canCreate) return;
   169→    if (isStaffUser && !selectedId) { setEquipment([]); setEquipLoaded(true); return; }
   170→    let alive = true;
   171→    setEquipLoading(true);
   172→    setEquipLoaded(false);
   173→    api.get<EquipmentOpt[]>(`/api/v1/equipment${qs({ customerId: selectedId || undefined, pageSize: 200 })}`)
   174→      .then((r) => {
   175→        if (!alive) return;
   176→        const list = Array.isArray(r.data) ? r.data : [];
   177→        setEquipment(list);
   178→        // Portal UX: derive the caller's own customer name from the equipment payload
   179→        // (customers lack customers.read by design — no extra API, no RBAC change).
   180→        if (!isStaffUser) {
   181→          const ownName = list.find((e) => e.customer?.companyName)?.customer?.companyName;
   182→          if (ownName) setPortalCustomerName(ownName);
   183→        }
   184→      })
   185→      .catch(() => { if (alive) setEquipment([]); })
   186→      .finally(() => { if (alive) { setEquipLoading(false); setEquipLoaded(true); } });
   187→    return () => { alive = false; };
   188→  }, [selectedId, isStaffUser, canCreate]);
   189→
   190→  const selectedEquipment = useMemo(
   191→    () => equipment.find((e) => e.id === draft.value.equipmentId) ?? null,
   192→    [equipment, draft.value.equipmentId]
   193→  );
   194→
   195→  // ── Dirty-state wiring (central router guard + data protection) ──
   196→  useEffect(() => {
   197→    setPageDirty(draft.dirty);
   198→    return () => { setPageDirty(false); };
   199→  }, [draft.dirty, setPageDirty]);
   200→
   201→  function goBackToList() {
   202→    draft.saveNow(); // silent protection — never lose typed data when leaving via Back
   203→    navigateTo("complaints");
   204→  }
   205→
   206→  function saveDraft() {
   207→    draft.saveNow();
   208→    toast({ title: "Draft saved successfully", description: "You can safely leave this page and restore the draft later." });
   209→  }
   210→
   211→  function pickCustomer(c: CustomerOpt) {
   212→    setSelectedCustomer(c);
   213→    setCustQuery(c.companyName ?? c.name ?? c.id);
   214→    setCustOpen(false);
   215→    draft.setValue({ customerId: c.id, equipmentId: "" });
   216→    setErrors((prev) => { const { customerId: _drop, ...rest } = prev; return rest; });
   217→  }
   218→
   219→  function clearCustomer() {
   220→    setSelectedCustomer(null);
   221→    setCustQuery("");
   222→    draft.setValue({ customerId: "", equipmentId: "" });
   223→  }
   224→
   225→  function validate(): Record<string, string> {
   226→    const errs: Record<string, string> = {};
   227→    const t = draft.value.title.trim();
   228→    if (!t) errs.title = "Title is required.";
   229→    else if (t.length < 3) errs.title = "Title must be at least 3 characters.";
   230→    const d = draft.value.description.trim();
   231→    if (!d) errs.description = "Description is required.";
   232→    else if (d.length < 3) errs.description = "Description must be at least 3 characters.";
   233→    if (isStaffUser && !draft.value.customerId) errs.customerId = "Please select a customer.";
   234→    return errs;
   235→  }
   236→
   237→  async function createComplaint() {
   238→    const errs = validate();
   239→    setErrors(errs);
   240→    if (Object.keys(errs).length > 0) {
   241→      toast({ title: "Check the highlighted fields", variant: "destructive" });
   242→      return;
   243→    }
   244→    setCreating(true);
   245→    setSubmitError(null);
   246→    try {
   247→      const payload: Record<string, unknown> = {
   248→        title: draft.value.title.trim(),
   249→        description: draft.value.description.trim(),
   250→        priority: draft.value.priority,
   251→        ...(isStaffUser && draft.value.customerId ? { customerId: draft.value.customerId } : {}),
   252→        ...(draft.value.equipmentId ? { equipmentId: draft.value.equipmentId } : {}),
   253→      };
   254→      const res = await api.post<{ id: string; code: string }>("/api/v1/complaints", payload);
   255→      toast({ title: "Complaint created", description: `${res.data.code} logged successfully.` });
   256→      draft.reset(EMPTY_CREATE);
   257→      setSelectedCustomer(null);
   258→      setCustQuery("");
   259→      setPageDirty(false);
   260→      // Continue the existing workflow on the complaint's dedicated detail page.
   261→      navigateTo("complaints", [res.data.id]);
   262→    } catch (e) {
   263→      // CRITICAL: keep every user-entered value on failure — show the error and allow retry.
   264→      const msg = e instanceof Error ? e.message : "Could not create the complaint. Please try again.";
   265→      setSubmitError(msg);
   266→      toast({ title: "Could not create complaint", description: msg, variant: "destructive" });
   267→    } finally {
   268→      setCreating(false);
   269→    }
   270→  }
   271→
   272→  if (!canCreate) {
   273→    return (
   274→      <EmptyState
   275→        title="You don't have permission to create complaints"
   276→        hint="Complaint creation is limited to authorized roles. Contact your administrator if you believe this is a mistake."
   277→      />
   278→    );
   279→  }
   280→
   281→  const actions = (
   282→    <>
   283→      <Button variant="outline" onClick={saveDraft} disabled={creating}>
   284→        <Save className="h-4 w-4 mr-1.5" /> Save Draft
   285→      </Button>
   286→      <Button onClick={createComplaint} disabled={creating}>
   287→        {creating ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> : <Send className="h-4 w-4 mr-1.5" />}
   288→        {creating ? "Creating…" : "Create Complaint"}
   289→      </Button>
   290→    </>
   291→  );
   292→
   293→  return (
   294→    <div>
   295→      {/* Breadcrumb */}
   296→      <nav aria-label="Breadcrumb" className="mb-2">
   297→        <ol className="flex items-center gap-1 text-sm text-muted-foreground">
   298→          <li>
   299→            <button
   300→              onClick={goBackToList}
   301→              className="hover:text-foreground hover:underline underline-offset-2 rounded"
   302→            >
   303→              Complaints
   304→            </button>
   305→          </li>
   306→          <li aria-hidden><ChevronRight className="h-3.5 w-3.5" /></li>
   307→          <li aria-current="page" className="text-foreground font-medium">New Complaint</li>
   308→        </ol>
   309→      </nav>
   310→
   311→      {/* Back link */}
   312→      <button
   313→        onClick={goBackToList}
   314→        className="mb-3 inline-flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors rounded"
   315→      >
   316→        <ArrowLeft className="h-4 w-4" aria-hidden /> Back to Complaints
   317→      </button>
   318→
   319→      <PageHeader
   320→        title="New Complaint"
   321→        subtitle={isStaffUser ? "Log a complaint on behalf of a customer." : "Report an issue — our team will assign a technician."}
   322→        actions={<div className="hidden sm:flex items-center gap-2 no-print">{actions}</div>}
   323→      />
   324→
   325→      {/* Recoverable draft banner */}
   326→      {draft.draftExists ? (
   327→        <div className="mb-4 flex flex-col sm:flex-row sm:items-center justify-between gap-2 rounded-lg border border-dashed bg-muted/30 p-3 text-sm">
   328→          <span className="text-muted-foreground">
   329→            Unsubmitted draft saved {draft.lastSavedAt ? fmtDateTime(draft.lastSavedAt) : "earlier"} — restore it to continue where you left off.
   330→          </span>
   331→          <div className="flex items-center gap-2">
   332→            <Button size="sm" variant="outline" onClick={draft.restore}>Restore</Button>
   333→            <Button size="sm" variant="ghost" onClick={draft.discard}>Discard</Button>
   334→          </div>
   335→        </div>
   336→      ) : null}
   337→
   338→      {submitError ? (
   339→        <div role="alert" className="mb-4 flex items-start gap-2 rounded-lg border border-destructive/40 bg-destructive/5 p-3 text-sm text-destructive">
   340→          <AlertCircle className="h-4 w-4 mt-0.5 shrink-0" aria-hidden />
   341→          <div>
   342→            <p className="font-medium">The complaint could not be created.</p>
   343→            <p className="mt-0.5">{submitError} Your entries are preserved — you can retry.</p>
   344→          </div>
   345→        </div>
   346→      ) : null}
   347→
   348→      <div className="grid gap-4 xl:grid-cols-5">
   349→        {/* ── Complaint information ── */}
   350→        <Card className="xl:col-span-3 shadow-sm">
   351→          <CardHeader className="pb-3">
   352→            <CardTitle className="text-base">Complaint Information</CardTitle>
   353→          </CardHeader>
   354→          <CardContent className="space-y-4">
   355→            <div className="space-y-1.5">
   356→              <div className="flex items-center justify-between">
   357→                <Label htmlFor="nc-title">Title *</Label>
   358→                <span className="text-[11px] text-muted-foreground tabular-nums">{draft.value.title.trim().length}/{TITLE_MAX}</span>
   359→              </div>
   360→              <Input
   361→                id="nc-title"
   362→                value={draft.value.title}
   363→                onChange={(e) => { draft.setValue({ title: e.target.value }); setErrors((p) => ({ ...p, title: "" })); }}
   364→                placeholder="Short summary of the issue — e.g. Air Conditioning Unit Not Cooling"
   365→                maxLength={TITLE_MAX}
   366→                aria-invalid={!!errors.title}
   367→                aria-describedby={errors.title ? "nc-title-err" : undefined}
   368→              />
   369→              {errors.title ? <p id="nc-title-err" className="text-xs text-destructive">{errors.title}</p> : null}
   370→            </div>
   371→
   372→            <div className="space-y-1.5">
   373→              <div className="flex items-center justify-between">
   374→                <Label htmlFor="nc-desc">Description *</Label>
   375→                <span className="text-[11px] text-muted-foreground tabular-nums">{draft.value.description.trim().length}/{DESC_MAX}</span>
   376→              </div>
   377→              <Textarea
   378→                id="nc-desc"
   379→                value={draft.value.description}
   380→                onChange={(e) => { draft.setValue({ description: e.target.value }); setErrors((p) => ({ ...p, description: "" })); }}
   381→                placeholder={"Describe the problem, location and impact…\n\nExample: Air conditioning unit in Level 2 meeting room is not cooling properly. Temperature remains above normal operating level."}
   382→                rows={7}
   383→                maxLength={DESC_MAX}
   384→                aria-invalid={!!errors.description}
   385→                aria-describedby={errors.description ? "nc-desc-err" : undefined}
   386→                className="min-h-[9rem]"
   387→              />
   388→              {errors.description ? <p id="nc-desc-err" className="text-xs text-destructive">{errors.description}</p> : null}
   389→            </div>
   390→
   391→            <div className="space-y-1.5 sm:max-w-xs">
   392→              <Label>Priority</Label>
   393→              <Select value={draft.value.priority} onValueChange={(v) => draft.setValue({ priority: v })}>
   394→                <SelectTrigger aria-label="Priority"><SelectValue /></SelectTrigger>
   395→                <SelectContent>
   396→                  {PRIORITIES.map((p) => <SelectItem key={p} value={p}>{humanize(p)}</SelectItem>)}
   397→                </SelectContent>
   398→              </Select>
   399→              <p className="text-[11px] text-muted-foreground">Defaults to Medium — the server enforces the same priority scale.</p>
   400→            </div>
   401→          </CardContent>
   402→        </Card>
   403→
   404→        {/* ── Customer & asset ── */}
   405→        <div className="xl:col-span-2 space-y-4">
   406→          <Card className="shadow-sm">
   407→            <CardHeader className="pb-3">
   408→              <CardTitle className="text-base">Customer &amp; Asset</CardTitle>
   409→            </CardHeader>
   410→            <CardContent className="space-y-4">
   411→              {isStaffUser ? (
   412→                <div className="space-y-1.5">
   413→                  <Label htmlFor="nc-customer">Customer *</Label>
   414→                  {selectedCustomer ? (
   415→                    <div className="flex items-start justify-between gap-2 rounded-lg border bg-muted/30 p-3">
   416→                      <div className="min-w-0 text-sm">
   417→                        <p className="font-medium truncate">{selectedCustomer.companyName ?? selectedCustomer.name ?? selectedCustomer.id}</p>
   418→                        <p className="text-xs text-muted-foreground truncate">
   419→                          {[selectedCustomer.contactPerson, selectedCustomer.phone, selectedCustomer.code ? `#${selectedCustomer.code}` : ""].filter(Boolean).join(" · ") || "—"}
   420→                        </p>
   421→                      </div>
   422→                      <Button size="icon" variant="ghost" className="h-7 w-7 shrink-0" onClick={clearCustomer} aria-label="Clear selected customer">
   423→                        <X className="h-3.5 w-3.5" />
   424→                      </Button>
   425→                    </div>
   426→                  ) : (
   427→                    <div className="relative">
   428→                      <Input
   429→                        id="nc-customer"
   430→                        role="combobox"
   431→                        aria-expanded={custOpen}
   432→                        aria-controls="nc-customer-list"
   433→                        aria-autocomplete="list"
   434→                        autoComplete="off"
   435→                        value={custQuery}
   436→                        onChange={(e) => { setCustQuery(e.target.value); setCustOpen(true); }}
   437→                        onFocus={() => setCustOpen(true)}
   438→                        onKeyDown={(e) => { if (e.key === "Escape") setCustOpen(false); }}
   439→                        placeholder="Search customer by name, contact or code…"
   440→                        aria-invalid={!!errors.customerId}
   441→                        aria-describedby={errors.customerId ? "nc-customer-err" : undefined}
   442→                      />
   443→                      {custLoading ? <Loader2 className="absolute right-3 top-2.5 h-4 w-4 animate-spin text-muted-foreground" aria-hidden /> : null}
   444→                      {custOpen ? (
   445→                        <>
   446→                          <div className="fixed inset-0 z-30" onClick={() => setCustOpen(false)} aria-hidden />
   447→                          <div
   448→                            id="nc-customer-list"
   449→                            role="listbox"
   450→                            aria-label="Customer results"
   451→                            className="absolute z-40 mt-1 w-full max-h-60 overflow-y-auto hms-scroll rounded-md border bg-background shadow-md"
   452→                          >
   453→                            {custLoading && custResults.length === 0 ? (
   454→                              <p className="px-3 py-4 text-sm text-muted-foreground flex items-center gap-2"><Loader2 className="h-3.5 w-3.5 animate-spin" /> Loading customers…</p>
   455→                            ) : custResults.length === 0 ? (
   456→                              <p className="px-3 py-4 text-sm text-muted-foreground">No customers match “{custQuery.trim()}”.</p>
   457→                            ) : (
   458→                              custResults.map((c) => (
   459→                                <button
   460→                                  key={c.id}
   461→                                  type="button"
   462→                                  role="option"
   463→                                  aria-selected={false}
   464→                                  onClick={() => pickCustomer(c)}
   465→                                  className="w-full text-left px-3 py-2 hover:bg-accent focus:bg-accent focus:outline-none border-b last:border-0"
   466→                                >
   467→                                  <span className="block text-sm font-medium truncate">{c.companyName ?? c.name ?? c.id}</span>
   468→                                  <span className="block text-xs text-muted-foreground truncate">
   469→                                    {[c.contactPerson, c.phone, c.code ? `#${c.code}` : ""].filter(Boolean).join(" · ") || "\u00A0"}
   470→                                  </span>
   471→                                </button>
   472→                              ))
   473→                            )}
   474→                          </div>
   475→                        </>

   491→              <div className="space-y-1.5">

   549→      {/* Autosave hint */}
