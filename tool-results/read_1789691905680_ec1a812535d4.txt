   142→   142→

   155→   155→          setSelectedCustomer(c);
   156→   156→          setCustQuery(c.companyName ?? c.name ?? c.id);
   157→   157→        }
   158→   158→      })
   159→   159→      .catch(() => undefined);
   160→   160→    return () => { alive = false; };
   161→   161→  }, [draft.value.customerId, isStaffUser, selectedCustomer?.id]);
   162→   162→
   163→   163→  // ── Equipment follows the selected customer (server-filtered) ──

   167→   167→
   168→   168→  useEffect(() => {
   169→   169→    if (!canCreate) return;

   177→   177→        const list = Array.isArray(r.data) ? r.data : [];

   184→   184→        }

   192→   192→    () => equipment.find((e) => e.id === draft.value.equipmentId) ?? null,

   205→   205→  }

   214→   214→    setCustQuery(c.companyName ?? c.name ?? c.id);

   226→   226→  function validate(): Record<string, string> {
   227→   227→    const errs: Record<string, string> = {};
   228→   228→    const t = draft.value.title.trim();
   229→   229→    if (!t) errs.title = "Title is required.";
   230→   230→    else if (t.length < 3) errs.title = "Title must be at least 3 characters.";
   231→   231→    const d = draft.value.description.trim();
   232→   232→    if (!d) errs.description = "Description is required.";
   233→   233→    else if (d.length < 3) errs.description = "Description must be at least 3 characters.";
   234→   234→    if (isStaffUser && !draft.value.customerId) errs.customerId = "Please select a customer.";
   235→   235→    return errs;
   236→   236→  }
   237→   237→
   238→   238→  async function createComplaint() {
   239→   239→    const errs = validate();
   240→   240→    setErrors(errs);
   241→   241→    if (Object.keys(errs).length > 0) {
   242→   242→      toast({ title: "Check the highlighted fields", variant: "destructive" });

   244→   244→    }
   245→   245→    setCreating(true);

   248→   248→      const payload: Record<string, unknown> = {
   249→   249→        title: draft.value.title.trim(),
   250→   250→        description: draft.value.description.trim(),
   251→   251→        priority: draft.value.priority,
   252→   252→        ...(isStaffUser && draft.value.customerId ? { customerId: draft.value.customerId } : {}),
   253→   253→        ...(draft.value.equipmentId ? { equipmentId: draft.value.equipmentId } : {}),
   254→   254→      };
   255→   255→      const res = await api.post<{ id: string; code: string }>("/api/v1/complaints", payload);
   256→   256→      toast({ title: "Complaint created", description: `${res.data.code} logged successfully.` });
   257→   257→      draft.reset(EMPTY_CREATE);
   258→   258→      setSelectedCustomer(null);
   259→   259→      setCustQuery("");
   260→   260→      setComplaintFormDirty(false);
   261→   261→      // Continue with the existing complaint workflow: open Complaint Details.
   262→   262→      setComplaintsFocusId(res.data.id);
   263→   263→      setComplaintsView("list");
   264→   264→    } catch (e) {
   265→   265→      // CRITICAL: keep every user-entered value on failure — show the error and allow retry.
   266→   266→      const msg = e instanceof Error ? e.message : "Could not create the complaint. Please try again.";
   267→   267→      setSubmitError(msg);
   268→   268→      toast({ title: "Could not create complaint", description: msg, variant: "destructive" });
   269→   269→    } finally {
   270→   270→      setCreating(false);
   271→   271→    }
   272→   272→  }
   273→   273→
   274→   274→  if (!canCreate) {
   275→   275→    return (
   276→   276→      <EmptyState
   277→   277→        title="You don't have permission to create complaints"
   278→   278→        hint="Complaint creation is limited to authorized roles. Contact your administrator if you believe this is a mistake."
   279→   279→      />
   280→   280→    );
   281→   281→  }

   283→   283→  const actions = (

   288→   288→      <Button onClick={createComplaint} disabled={creating}>

   291→   291→      </Button>
   292→   292→    </>

   297→   297→      {/* Breadcrumb */}
   298→   298→      <nav aria-label="Breadcrumb" className="mb-2">
   299→   299→        <ol className="flex items-center gap-1 text-sm text-muted-foreground">
   300→   300→          <li>
   301→   301→            <button
   302→   302→              onClick={goBackToList}
   303→   303→              className="hover:text-foreground hover:underline underline-offset-2 rounded"
   304→   304→            >
   305→   305→              Complaints
   306→   306→            </button>
   307→   307→          </li>
   308→   308→          <li aria-hidden><ChevronRight className="h-3.5 w-3.5" /></li>

   311→   311→      </nav>

   313→   313→      {/* Back link */}

   316→   316→        className="mb-3 inline-flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors rounded"

   318→   318→        <ArrowLeft className="h-4 w-4" aria-hidden /> Back to Complaints

   320→   320→
   321→   321→      <PageHeader
   322→   322→        title="New Complaint"
   323→   323→        subtitle={isStaffUser ? "Log a complaint on behalf of a customer." : "Report an issue — our team will assign a technician."}
   324→   324→        actions={<div className="hidden sm:flex items-center gap-2 no-print">{actions}</div>}

   326→   326→
   327→   327→      {/* Recoverable draft banner */}

   329→   329→        <div className="mb-4 flex flex-col sm:flex-row sm:items-center justify-between gap-2 rounded-lg border border-dashed bg-muted/30 p-3 text-sm">

   331→   331→            Unsubmitted draft saved {draft.lastSavedAt ? fmtDateTime(draft.lastSavedAt) : "earlier"} — restore it to continue where you left off.

   333→   333→          <div className="flex items-center gap-2">
   334→   334→            <Button size="sm" variant="outline" onClick={draft.restore}>Restore</Button>
   335→   335→            <Button size="sm" variant="ghost" onClick={draft.discard}>Discard</Button>
   336→   336→          </div>
   337→   337→        </div>
   338→   338→      ) : null}
   339→   339→
   340→   340→      {submitError ? (
   341→   341→        <div role="alert" className="mb-4 flex items-start gap-2 rounded-lg border border-destructive/40 bg-destructive/5 p-3 text-sm text-destructive">
   342→   342→          <AlertCircle className="h-4 w-4 mt-0.5 shrink-0" aria-hidden />
   343→   343→          <div>
   344→   344→            <p className="font-medium">The complaint could not be created.</p>
   345→   345→            <p className="mt-0.5">{submitError} Your entries are preserved — you can retry.</p>

   347→   347→        </div>

   349→   349→
   350→   350→      <div className="grid gap-4 xl:grid-cols-5">
   351→   351→        {/* ── Complaint information ── */}
   352→   352→        <Card className="xl:col-span-3 shadow-sm">

   354→   354→            <CardTitle className="text-base">Complaint Information</CardTitle>
   355→   355→          </CardHeader>
   356→   356→          <CardContent className="space-y-4">

   358→   358→              <div className="flex items-center justify-between">
   359→   359→                <Label htmlFor="nc-title">Title *</Label>

   371→   371→              {errors.title ? <p id="nc-title-err" className="text-xs text-destructive">{errors.title}</p> : null}
   372→   372→            </div>
   373→   373→
   374→   374→            <div className="space-y-1.5">
   375→   375→              <div className="flex items-center justify-between">
   376→   376→                <Label htmlFor="nc-desc">Description *</Label>
   377→   377→                <span className="text-[11px] text-muted-foreground tabular-nums">{draft.value.description.trim().length}/{DESC_MAX}</span>
   378→   378→              </div>

   380→   380→                id="nc-desc"

   384→   384→                rows={7}
   385→   385→                maxLength={DESC_MAX}
   386→   386→                aria-invalid={!!errors.description}
   387→   387→                aria-describedby={errors.description ? "nc-desc-err" : undefined}
   388→   388→                className="min-h-[9rem]"
   389→   389→              />
   390→   390→              {errors.description ? <p id="nc-desc-err" className="text-xs text-destructive">{errors.description}</p> : null}
   391→   391→            </div>
   392→   392→
   393→   393→            <div className="space-y-1.5 sm:max-w-xs">
   394→   394→              <Label>Priority</Label>
   395→   395→              <Select value={draft.value.priority} onValueChange={(v) => draft.setValue({ priority: v })}>
   396→   396→                <SelectTrigger aria-label="Priority"><SelectValue /></SelectTrigger>
   397→   397→                <SelectContent>
   398→   398→                  {PRIORITIES.map((p) => <SelectItem key={p} value={p}>{humanize(p)}</SelectItem>)}
   399→   399→                </SelectContent>
   400→   400→              </Select>
   401→   401→              <p className="text-[11px] text-muted-foreground">Defaults to Medium — the server enforces the same priority scale.</p>
   402→   402→            </div>
   403→   403→          </CardContent>
   404→   404→        </Card>
   405→   405→
   406→   406→        {/* ── Customer & asset ── */}
   407→   407→        <div className="xl:col-span-2 space-y-4">
