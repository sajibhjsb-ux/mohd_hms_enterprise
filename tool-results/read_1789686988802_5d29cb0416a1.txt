     8→     8→import { useCallback, useEffect, useMemo, useState } from "react";
     9→     9→import { api, qs } from "@/lib/hms/api-client";
    10→    10→import { hasPerm, useSession } from "@/components/hms/session";
    11→    11→import { useToast } from "@/hooks/use-toast";
    12→    12→import { useDraft } from "@/hooks/use-draft";
    13→    13→import { DataTable, type Column } from "@/components/hms/shared/data-table";
    14→    14→import {
    15→    15→  PageHeader, StatCard, StatusBadge, PriorityBadge, LoadingState, EmptyState, ErrorState,
    16→    16→} from "@/components/hms/shared/ui-bits";
    17→    17→import { PERMISSIONS, PRIORITIES, humanize } from "@/lib/hms/constants";
    18→    18→import { fmtDate, fmtDateTime } from "@/lib/hms/format";
    19→    19→import { Button } from "@/components/ui/button";
    20→    20→import { Input } from "@/components/ui/input";
    21→    21→import { Label } from "@/components/ui/label";
    22→    22→import { Textarea } from "@/components/ui/textarea";
    23→    23→import { Badge } from "@/components/ui/badge";
    24→    24→import { Separator } from "@/components/ui/separator";
    25→    25→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    26→    26→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    27→    27→import {
    28→    28→  AlertTriangle, CheckCircle2, ClipboardCheck, Clock, Hammer, ListChecks, Plus, Send,
    29→    29→} from "lucide-react";

    32→    32→// ── Types ──
    33→    33→
    34→    34→type ComplaintRow = {
    35→    35→  id: string;
    36→    36→  code: string;
    37→    37→  title: string;
    38→    38→  description: string;
    39→    39→  priority: string;
    40→    40→  status: string;
    41→    41→  createdAt: string;
    42→    42→  customerId: string;
    43→    43→  customer?: { id: string; companyName: string } | null;
    44→    44→  equipment?: { id: string; name: string; assetTag: string } | null;
    45→    45→  assignedTechnician?: { id: string; user?: { id: string; name: string } | null } | null;
    46→    46→};
    47→    47→
    48→    48→type HistoryRow = {
    49→    49→  id: string;
    50→    50→  fromStatus: string;
    51→    51→  toStatus: string;
    52→    52→  note: string;
    53→    53→  createdAt: string;
    54→    54→  changedByName?: string | null;
    55→    55→};
    56→    56→
    57→    57→type ComplaintDetail = ComplaintRow & {
    58→    58→  resolutionNotes: string;
    59→    59→  customerFeedback: string;
    60→    60→  assignedAt: string | null;
    61→    61→  acceptedAt: string | null;
    62→    62→  startedAt: string | null;
    63→    63→  completedAt: string | null;
    64→    64→  confirmedAt: string | null;
    65→    65→  closedAt: string | null;
    66→    66→  statusHistory: HistoryRow[];
    67→    67→  workOrders: { id: string; code: string; title: string; status: string; technician?: { user?: { name: string } | null } | null }[];
    68→    68→};
    69→    69→
    70→    70→type TechOpt = { id: string; employeeNo?: string; specialty?: string; user?: { name?: string } | null };
    71→    71→type CustomerOpt = { id: string; companyName?: string; name?: string };
    72→    72→type EquipmentOpt = { id: string; name?: string; assetTag?: string };
    73→    73→
    74→    74→type CreateForm = {
    75→    75→  title: string;
    76→    76→  description: string;
    77→    77→  priority: string;
    78→    78→  customerId: string;
    79→    79→  equipmentId: string;
    80→    80→};
    81→    81→
    82→    82→const EMPTY_CREATE: CreateForm = { title: "", description: "", priority: "MEDIUM", customerId: "", equipmentId: "" };
    83→    83→
    84→    84→const OPEN_STATUSES = ["NEW", "ASSIGNED"];
    85→    85→const PROGRESS_STATUSES = ["IN_PROGRESS"];
    86→    86→const RESOLVED_STATUSES = ["COMPLETED", "CONFIRMED"];
    87→    87→
    88→    88→const STATUS_TABS: { key: string; label: string; match: (s: string) => boolean }[] = [
    89→    89→  { key: "ALL", label: "All", match: () => true },
    90→    90→  { key: "OPEN", label: "Open", match: (s) => OPEN_STATUSES.includes(s) },
    91→    91→  { key: "IN_PROGRESS", label: "In Progress", match: (s) => PROGRESS_STATUSES.includes(s) },
    92→    92→  { key: "RESOLVED", label: "Resolved", match: (s) => RESOLVED_STATUSES.includes(s) },
    93→    93→  { key: "CLOSED", label: "Closed", match: (s) => s === "CLOSED" },
    94→    94→  { key: "CANCELLED", label: "Cancelled", match: (s) => s === "CANCELLED" },
    95→    95→];
    96→    96→
    97→    97→// ── Module ──
    98→    98→
    99→    99→export function ComplaintsModule() {
   100→   100→  const { user } = useSession();
   101→   101→  const { toast } = useToast();
   102→   102→
   103→   103→  const canCreate = hasPerm(user, PERMISSIONS.complaints_create);
   104→   104→  const canAssign = hasPerm(user, PERMISSIONS.complaints_assign);
   105→   105→  const canUpdate = hasPerm(user, PERMISSIONS.complaints_update);
   106→   106→  const canClose = hasPerm(user, PERMISSIONS.complaints_close);
   107→   107→  const isStaffUser = !!user && user.role !== "CUSTOMER";
   108→   108→
   109→   109→  // ── List state ──
   110→   110→  const [rows, setRows] = useState<ComplaintRow[] | null>(null);
   111→   111→  const [loadError, setLoadError] = useState<string | null>(null);
   112→   112→  const [loading, setLoading] = useState(true);
   113→   113→  const [tab, setTab] = useState("ALL");
   114→   114→  const [reloadKey, setReloadKey] = useState(0);
   115→   115→
   116→   116→  const load = useCallback(async () => {
   117→   117→    setLoading(true);
   118→   118→    setLoadError(null);
   119→   119→    try {
   120→   120→      const res = await api.get<ComplaintRow[]>(`/api/v1/complaints${qs({ pageSize: 200 })}`);
   121→   121→      setRows(res.data);
   122→   122→    } catch (e) {
   123→   123→      setLoadError(e instanceof Error ? e.message : "Failed to load complaints.");
   124→   124→    } finally {
   125→   125→      setLoading(false);
   126→   126→    }
   127→   127→  }, []);
   128→   128→
   129→   129→  useEffect(() => { load(); }, [load, reloadKey]);
   130→   130→
   131→   131→  // ── Detail state ──
   132→   132→  const [detail, setDetail] = useState<ComplaintDetail | null>(null);
   133→   133→  const [detailOpen, setDetailOpen] = useState(false);
   134→   134→  const [detailLoading, setDetailLoading] = useState(false);
   135→   135→  const [note, setNote] = useState("");
   136→   136→  const [busy, setBusy] = useState(false);
   137→   137→
   138→   138→  const [techs, setTechs] = useState<TechOpt[]>([]);
   139→   139→  const [assignTo, setAssignTo] = useState<string>("");
   140→   140→
   141→   141→  const openDetail = useCallback(async (id: string) => {
   142→   142→    setDetailOpen(true);
   143→   143→    setDetailLoading(true);
   144→   144→    setDetail(null);
   145→   145→    setNote("");
   146→   146→    setAssignTo("");
   147→   147→    try {
   148→   148→      const res = await api.get<ComplaintDetail>(`/api/v1/complaints/${id}`);
   149→   149→      setDetail(res.data);
   150→   150→    } catch (e) {
   151→   151→      toast({ title: "Could not load complaint", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   152→   152→      setDetailOpen(false);
   153→   153→    } finally {
   154→   154→      setDetailLoading(false);
   155→   155→    }
   156→   156→  }, [toast]);
   157→   157→
   158→   158→  // Technician options for the assign control
   159→   159→  useEffect(() => {
   160→   160→    if (!detailOpen || !canAssign) return;
   161→   161→    let alive = true;
   162→   162→    api.get<TechOpt[]>(`/api/v1/technicians${qs({ pageSize: 200 })}`)
   163→   163→      .then((r) => { if (alive) setTechs(Array.isArray(r.data) ? r.data : []); })
   164→   164→      .catch(() => { if (alive) setTechs([]); });
   165→   165→    return () => { alive = false; };
   166→   166→  }, [detailOpen, canAssign]);
   167→   167→
   168→   168→  const runTransition = useCallback(async (action: string, extra?: Record<string, unknown>) => {
   169→   169→    if (!detail) return;
   170→   170→    setBusy(true);
   171→   171→    try {
   172→   172→      await api.post<ComplaintDetail>(`/api/v1/complaints/${detail.id}/transition`, { action, ...extra });
   173→   173→      toast({ title: "Success", description: `Complaint ${detail.code} — ${humanize(action)} done.` });
   174→   174→      setDetailOpen(false);
   175→   175→      setDetail(null);
   176→   176→      setReloadKey((k) => k + 1);
   177→   177→    } catch (e) {
   178→   178→      toast({ title: "Action failed", description: e instanceof Error ? e.message : "Please try again.", variant: "destructive" });
   179→   179→    } finally {
   180→   180→      setBusy(false);
   181→   181→    }
   182→   182→  }, [detail, toast]);
   183→   183→
   184→   184→  // ── Create dialog ──
   185→   185→  const [createOpen, setCreateOpen] = useState(false);
   186→   186→  const [submitting, setSubmitting] = useState(false);
   187→   187→  const [customers, setCustomers] = useState<CustomerOpt[]>([]);
   188→   188→  const [equipment, setEquipment] = useState<EquipmentOpt[]>([]);
   189→   189→  const draft = useDraft<CreateForm>({ formKey: "complaint.create", initial: EMPTY_CREATE });
   190→   190→
   191→   191→  useEffect(() => {
   192→   192→    if (!createOpen || !isStaffUser) return;
   193→   193→    let alive = true;
   194→   194→    api.get<CustomerOpt[]>(`/api/v1/customers${qs({ pageSize: 200 })}`)
   195→   195→      .then((r) => { if (alive) setCustomers(Array.isArray(r.data) ? r.data : []); })
   196→   196→      .catch(() => { if (alive) setCustomers([]); });
   197→   197→    return () => { alive = false; };
   198→   198→  }, [createOpen, isStaffUser]);
   199→   199→
   200→   200→  // Equipment options follow the selected customer (or the caller's own scope for portal users)
   201→   201→  useEffect(() => {
   202→   202→    if (!createOpen) return;
   203→   203→    if (isStaffUser && !draft.value.customerId) { setEquipment([]); return; }
   204→   204→    let alive = true;
   205→   205→    api.get<EquipmentOpt[]>(`/api/v1/equipment${qs({ customerId: draft.value.customerId || undefined, pageSize: 200 })}`)
   206→   206→      .then((r) => { if (alive) setEquipment(Array.isArray(r.data) ? r.data : []); })
   207→   207→      .catch(() => { if (alive) setEquipment([]); });
   208→   208→    return () => { alive = false; };
   209→   209→  }, [createOpen, draft.value.customerId, isStaffUser]);
   210→   210→
   211→   211→  async function submitCreate() {
   212→   212→    const v = draft.value;
   213→   213→    if (v.title.trim().length < 3 || v.description.trim().length < 3) {
   214→   214→      toast({ title: "Check the form", description: "Title and description need at least 3 characters.", variant: "destructive" });
   215→   215→      return;
   216→   216→    }
   217→   217→    setSubmitting(true);
   218→   218→    try {
   219→   219→      const payload: Record<string, unknown> = {
   220→   220→        title: v.title.trim(),
   221→   221→        description: v.description.trim(),
   222→   222→        priority: v.priority,
   223→   223→        ...(isStaffUser && v.customerId ? { customerId: v.customerId } : {}),
   224→   224→        ...(v.equipmentId ? { equipmentId: v.equipmentId } : {}),
   225→   225→      };
   226→   226→      const res = await api.post<ComplaintRow>("/api/v1/complaints", payload);
   227→   227→      toast({ title: "Complaint created", description: `${res.data.code} logged successfully.` });
   228→   228→      draft.reset(EMPTY_CREATE);
   229→   229→      setCreateOpen(false);
   230→   230→      setReloadKey((k) => k + 1);
   231→   231→    } catch (e) {
   232→   232→      toast({ title: "Could not create complaint", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   233→   233→    } finally {
   234→   234→      setSubmitting(false);
   235→   235→    }
   236→   236→  }
   237→   237→
   238→   238→  // ── Derived views ──
   239→   239→  const stats = useMemo(() => {
   240→   240→    const list = rows ?? [];
   241→   241→    const count = (match: (r: ComplaintRow) => boolean) => list.filter(match).length;
   242→   242→    return {
   243→   243→      total: list.length,
   244→   244→      open: count((r) => OPEN_STATUSES.includes(r.status)),
   245→   245→      inProgress: count((r) => PROGRESS_STATUSES.includes(r.status)),
   246→   246→      resolved: count((r) => RESOLVED_STATUSES.includes(r.status)),
   247→   247→      closed: count((r) => r.status === "CLOSED"),
   248→   248→      urgent: count((r) => r.priority === "URGENT" && !["CLOSED", "CANCELLED"].includes(r.status)),
   249→   249→    };
   250→   250→  }, [rows]);
   251→   251→
   252→   252→  const visibleRows = useMemo(() => {
   253→   253→    const active = STATUS_TABS.find((t) => t.key === tab) ?? STATUS_TABS[0];
   254→   254→    return (rows ?? []).filter((r) => active.match(r.status));
   255→   255→  }, [rows, tab]);
   256→   256→
   257→   257→  const tabCount = (key: string) => {
   258→   258→    const active = STATUS_TABS.find((t) => t.key === key);
   259→   259→    if (!active) return 0;
   260→   260→    return (rows ?? []).filter((r) => active.match(r.status)).length;
   261→   261→  };
   262→   262→
   263→   263→  const columns: Column<ComplaintRow>[] = [
   264→   264→    { key: "code", header: "Code", value: (r) => r.code, className: "font-mono text-xs whitespace-nowrap" },
   265→   265→    { key: "title", header: "Title", value: (r) => r.title, className: "max-w-[260px] truncate" },
   266→   266→    { key: "priority", header: "Priority", render: (r) => <PriorityBadge priority={r.priority} />, value: (r) => r.priority },
   267→   267→    { key: "status", header: "Status", render: (r) => <StatusBadge status={r.status} />, value: (r) => r.status },
   268→   268→    { key: "customer", header: "Customer", value: (r) => r.customer?.companyName ?? "", hideOnMobile: true },
   269→   269→    { key: "technician", header: "Technician", value: (r) => r.assignedTechnician?.user?.name ?? "", render: (r) => r.assignedTechnician?.user?.name ?? "—", hideOnMobile: true },
   270→   270→    { key: "equipment", header: "Equipment", value: (r) => r.equipment?.name ?? "", render: (r) => (r.equipment ? `${r.equipment.name} (${r.equipment.assetTag})` : "—"), hideOnMobile: true },
   271→   271→    { key: "createdAt", header: "Created", value: (r) => r.createdAt, render: (r) => fmtDate(r.createdAt), hideOnMobile: true },
   272→   272→  ];
   273→   273→
   274→   274→  // ── Detail action visibility ──
   275→   275→  const isAssignedTech = !!detail && !!user && detail.assignedTechnician?.user?.id === user.id;
   276→   276→  const isPortalOwner = !!detail && !!user && user.role === "CUSTOMER" && user.customerId === detail.customerId;
   277→   277→  const status = detail?.status;
   278→   278→
   279→   279→  return (
   280→   280→    <div>
   281→   281→      <PageHeader
   282→   282→        title="Complaints"
   283→   283→        subtitle={isStaffUser ? "Track, assign and resolve customer complaints end-to-end." : "Your complaints and their live progress."}
   284→   284→        actions={canCreate ? (
   285→   285→          <Button onClick={() => { setCreateOpen(true); }}>
   286→   286→            <Plus className="h-4 w-4 mr-1.5" /> New Complaint
   287→   287→          </Button>
   288→   288→        ) : null}
   289→   289→      />
   290→   290→
   291→   291→      {/* Stats */}
   292→   292→      <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-6 gap-3 mb-5">
   293→   293→        <StatCard title="Total" value={stats.total} icon={<ListChecks className="h-5 w-5" />} loading={!rows && loading} />
   294→   294→        <StatCard title="Open" value={stats.open} icon={<Clock className="h-5 w-5" />} tone="warning" loading={!rows && loading} />
   295→   295→        <StatCard title="In Progress" value={stats.inProgress} icon={<Hammer className="h-5 w-5" />} tone="warning" loading={!rows && loading} />
   296→   296→        <StatCard title="Resolved" value={stats.resolved} icon={<CheckCircle2 className="h-5 w-5" />} tone="success" loading={!rows && loading} />
   297→   297→        <StatCard title="Closed" value={stats.closed} icon={<ClipboardCheck className="h-5 w-5" />} loading={!rows && loading} />
   298→   298→        <StatCard title="Urgent Active" value={stats.urgent} icon={<AlertTriangle className="h-5 w-5" />} tone="danger" loading={!rows && loading} />
   299→   299→      </div>
   300→   300→
   301→   301→      {/* Status tabs */}
   302→   302→      <div className="flex flex-wrap gap-1.5 mb-4">
   303→   303→        {STATUS_TABS.map((t) => (
   304→   304→          <button
   305→   305→            key={t.key}
   306→   306→            onClick={() => setTab(t.key)}
   307→   307→            className={cn(
   308→   308→              "px-3 py-1.5 rounded-full text-xs font-medium border transition-colors",
   309→   309→              tab === t.key ? "bg-primary text-primary-foreground border-primary" : "bg-background hover:bg-muted text-muted-foreground"
   310→   310→            )}
   311→   311→          >
   312→   312→            {t.label} <span className="opacity-70 tabular-nums">({tabCount(t.key)})</span>
   313→   313→          </button>
   314→   314→        ))}
   315→   315→      </div>
   316→   316→
   317→   317→      {loading && !rows ? (
   318→   318→        <LoadingState label="Loading complaints…" rows={5} />
   319→   319→      ) : loadError ? (
   320→   320→        <ErrorState message={loadError} onRetry={load} />
   321→   321→      ) : visibleRows.length === 0 ? (
   322→   322→        <EmptyState
   323→   323→          title="No complaints in this view"
   324→   324→          hint={canCreate ? "Log a new complaint to get started — drafts are saved automatically while you type." : "Complaints will appear here as they are filed."}
   325→   325→          action={canCreate ? <Button variant="outline" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1.5" /> New Complaint</Button> : undefined}
   326→   326→        />
   327→   327→      ) : (
   328→   328→        <DataTable
   329→   329→          columns={columns}
   330→   330→          rows={visibleRows}
   331→   331→          rowKey={(r) => r.id}
   332→   332→          onRowClick={(r) => openDetail(r.id)}
   333→   333→          searchPlaceholder="Search code, title, description…"
   334→   334→          filters={[{
   335→   335→            key: "priority",
   336→   336→            label: "Priorities",
   337→   337→            options: PRIORITIES.map((p) => ({ value: p, label: humanize(p) })),
   338→   338→            match: (row, value) => row.priority === value,
   339→   339→          }]}
   340→   340→          emptyTitle="No complaints match"
   341→   341→          exportName="complaints"
   342→   342→        />
   343→   343→      )}
   344→   344→
   345→   345→      {/* ── Detail dialog ── */}
   346→   346→      <Dialog open={detailOpen} onOpenChange={(open) => { if (!open) { setDetailOpen(false); setDetail(null); } }}>
   347→   347→        <DialogContent className="max-w-3xl max-h-[88vh] overflow-y-auto hms-scroll">
   348→   348→          <DialogTitle className="sr-only">Details</DialogTitle>
   349→   349→          {detailLoading || !detail ? (
   350→   350→            <LoadingState label="Loading complaint…" />
   351→   351→          ) : (
   352→   352→            <>
   353→   353→              <DialogHeader>
   354→   354→                <DialogTitle className="flex flex-wrap items-center gap-2">
   355→   355→                  <span className="font-mono text-sm text-muted-foreground">{detail.code}</span>
   356→   356→                  <span className="text-base">{detail.title}</span>
   357→   357→                  <StatusBadge status={detail.status} />
   358→   358→                  <PriorityBadge priority={detail.priority} />
   359→   359→                </DialogTitle>
   360→   360→                <DialogDescription>
   361→   361→                  {detail.customer?.companyName ?? "—"}
   362→   362→                  {detail.equipment ? ` · ${detail.equipment.name} (${detail.equipment.assetTag})` : ""}
   363→   363→                  {` · Logged ${fmtDateTime(detail.createdAt)}`}
   364→   364→                </DialogDescription>
   365→   365→              </DialogHeader>
   366→   366→
   367→   367→              <div className="grid gap-4 sm:grid-cols-2 text-sm">
   368→   368→                <div className="space-y-1">
   369→   369→                  <p className="text-muted-foreground text-xs uppercase tracking-wide">Technician</p>
   370→   370→                  <p>{detail.assignedTechnician?.user?.name ?? "Unassigned"}</p>
   371→   371→                </div>
   372→   372→                <div className="space-y-1">
   373→   373→                  <p className="text-muted-foreground text-xs uppercase tracking-wide">Reported</p>
   374→   374→                  <p>{fmtDateTime(detail.createdAt)}</p>
   375→   375→                </div>
   376→   376→              </div>
   377→   377→
   378→   378→              <div className="text-sm space-y-1">
   379→   379→                <p className="text-muted-foreground text-xs uppercase tracking-wide">Description</p>
   380→   380→                <p className="whitespace-pre-wrap rounded-lg bg-muted/40 p-3">{detail.description}</p>
   381→   381→              </div>
   382→   382→
   383→   383→              {detail.resolutionNotes ? (
   384→   384→                <div className="text-sm space-y-1">
   385→   385→                  <p className="text-muted-foreground text-xs uppercase tracking-wide">Resolution notes</p>
   386→   386→                  <p className="whitespace-pre-wrap rounded-lg bg-muted/40 p-3">{detail.resolutionNotes}</p>
   387→   387→                </div>
   388→   388→              ) : null}
   389→   389→              {detail.customerFeedback ? (
   390→   390→                <div className="text-sm space-y-1">
