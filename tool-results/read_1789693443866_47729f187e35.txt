     1→"use client";

     3→// MOHD.HMS ENTERPRISE — Work Orders module.
     4→// PENDING → ACCEPTED → IN_PROGRESS → COMPLETED (+ ON_HOLD, CANCELLED).
     5→// Checklist execution, materials with live totals, labour cost capture,
     6→// role-gated transitions. Draft-protected creation form.
     7→
     8→import { useCallback, useEffect, useMemo, useState } from "react";
     9→import { api, qs } from "@/lib/hms/api-client";
    10→import { hasPerm, useSession } from "@/components/hms/session";
    11→import { useToast } from "@/hooks/use-toast";
    12→import { useDraft } from "@/hooks/use-draft";
    13→import { DataTable, type Column } from "@/components/hms/shared/data-table";
    14→import {
    15→  PageHeader, StatCard, StatusBadge, PriorityBadge, LoadingState, EmptyState, ErrorState,
    16→} from "@/components/hms/shared/ui-bits";
    17→import { PERMISSIONS, PRIORITIES, humanize } from "@/lib/hms/constants";
    18→import { money, fmtDate, fmtDateTime, toCents } from "@/lib/hms/format";
    19→import { Button } from "@/components/ui/button";
    20→import { Input } from "@/components/ui/input";
    21→import { Label } from "@/components/ui/label";
    22→import { Textarea } from "@/components/ui/textarea";
    23→import { Badge } from "@/components/ui/badge";
    24→import { Checkbox } from "@/components/ui/checkbox";
    25→import { Separator } from "@/components/ui/separator";
    26→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    27→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    28→import {
    29→  CircleDollarSign, ClipboardList, Hammer, PauseCircle, PlayCircle, Plus, Trash2, Wrench,
    30→} from "lucide-react";
    31→import { cn } from "@/lib/utils";
    32→
    33→// ── Types ──
    34→
    35→type WORow = {
    36→  id: string;
    37→  code: string;
    38→  title: string;
    39→  status: string;
    40→  priority: string;
    41→  scheduledDate: string | null;
    42→  totalCents: number;
    43→  createdAt: string;
    44→  customer?: { id: string; companyName: string } | null;
    45→  equipment?: { id: string; name: string; assetTag: string } | null;
    46→  technician?: { id: string; user?: { id: string; name: string } | null } | null;
    47→  complaint?: { id: string; code: string } | null;
    48→};
    49→
    50→type ChecklistItem = { id: string; label: string; done: boolean; doneAt: string | null; sortOrder: number };
    51→type MaterialRow = {
    52→  id: string; name: string; quantity: number; unit: string; unitCostCents: number; totalCents: number;
    53→  inventoryItemId: string | null; inventoryItem?: { id: string; name: string; unit: string } | null;
    54→};
    55→
    56→type WODetail = WORow & {
    57→  description: string;
    58→  notes: string;
    59→  customerId: string;
    60→  startedAt: string | null;
    61→  completedAt: string | null;
    62→  labourHours: number;
    63→  labourRateCents: number;
    64→  labourTotalCents: number;
    65→  materialsTotalCents: number;
    66→  checklist: ChecklistItem[];
    67→  materials: MaterialRow[];
    68→};
    69→
    70→type TechOpt = { id: string; employeeNo?: string; specialty?: string; user?: { name?: string } | null };
    71→type CustomerOpt = { id: string; companyName?: string; name?: string };
    72→type EquipmentOpt = { id: string; name?: string; assetTag?: string };
    73→type ComplaintOpt = { id: string; code: string; title: string; customer?: { id: string; companyName?: string } | null };
    74→
    75→type WOMaterialForm = { name: string; qty: string; cost: string };
    76→type WOCreateForm = {
    77→  title: string;
    78→  description: string;
    79→  customerId: string;
    80→  equipmentId: string;
    81→  complaintId: string;
    82→  technicianId: string;
    83→  priority: string;
    84→  scheduledDate: string;
    85→  checklistText: string;
    86→  materials: WOMaterialForm[];
    87→};
    88→
    89→const EMPTY_CREATE: WOCreateForm = {
    90→  title: "", description: "", customerId: "", equipmentId: "", complaintId: "",
    91→  technicianId: "", priority: "MEDIUM", scheduledDate: "", checklistText: "", materials: [],
    92→};
    93→
    94→const PENDING_STATUSES = ["PENDING", "ACCEPTED"];
    95→
    96→const STATUS_TABS: { key: string; label: string; match: (s: string) => boolean }[] = [
    97→  { key: "ALL", label: "All", match: () => true },
    98→  { key: "PENDING", label: "Pending", match: (s) => PENDING_STATUSES.includes(s) },
    99→  { key: "IN_PROGRESS", label: "In Progress", match: (s) => s === "IN_PROGRESS" },
   100→  { key: "ON_HOLD", label: "On Hold", match: (s) => s === "ON_HOLD" },
   101→  { key: "COMPLETED", label: "Completed", match: (s) => s === "COMPLETED" },
   102→  { key: "CANCELLED", label: "Cancelled", match: (s) => s === "CANCELLED" },
   103→];
   104→
   105→export function WorkOrdersModule() {
   106→  const { user } = useSession();
   107→  const { toast } = useToast();
   108→
   109→  const canCreate = hasPerm(user, PERMISSIONS.work_orders_create);
   110→  const canAssign = hasPerm(user, PERMISSIONS.work_orders_assign);
   111→  const canUpdate = hasPerm(user, PERMISSIONS.work_orders_update);
   112→  const canComplete = hasPerm(user, PERMISSIONS.work_orders_complete);
   113→  const isStaffUser = !!user && user.role !== "CUSTOMER";
   114→
   115→  // ── List ──
   116→  const [rows, setRows] = useState<WORow[] | null>(null);
   117→  const [loadError, setLoadError] = useState<string | null>(null);
   118→  const [loading, setLoading] = useState(true);
   119→  const [tab, setTab] = useState("ALL");
   120→  const [reloadKey, setReloadKey] = useState(0);
   121→
   122→  const load = useCallback(async () => {
   123→    setLoading(true);
   124→    setLoadError(null);
   125→    try {
   126→      const res = await api.get<WORow[]>(`/api/v1/work-orders${qs({ pageSize: 200 })}`);
   127→      setRows(res.data);
   128→    } catch (e) {
   129→      setLoadError(e instanceof Error ? e.message : "Failed to load work orders.");
   130→    } finally {
   131→      setLoading(false);
   132→    }
   133→  }, []);
   134→
   135→  useEffect(() => { load(); }, [load, reloadKey]);
   136→
   137→  // ── Detail ──
   138→  const [detail, setDetail] = useState<WODetail | null>(null);
   139→  const [detailOpen, setDetailOpen] = useState(false);
   140→  const [detailLoading, setDetailLoading] = useState(false);
   141→  const [busy, setBusy] = useState(false);
   142→  const [completeNote, setCompleteNote] = useState("");
   143→  const [newItem, setNewItem] = useState("");
   144→  const [matForm, setMatForm] = useState<WOMaterialForm>({ name: "", qty: "1", cost: "0" });
   145→  const [labourHours, setLabourHours] = useState("0");
   146→  const [labourRate, setLabourRate] = useState("0");
   147→
   148→  const openDetail = useCallback(async (id: string) => {
   149→    setDetailOpen(true);
   150→    setDetailLoading(true);
   151→    setDetail(null);
   152→    setCompleteNote("");
   153→    setNewItem("");
   154→    setMatForm({ name: "", qty: "1", cost: "0" });
   155→    try {
   156→      const res = await api.get<WODetail>(`/api/v1/work-orders/${id}`);
   157→      setDetail(res.data);
   158→      setLabourHours(String(res.data.labourHours ?? 0));
   159→      setLabourRate(((res.data.labourRateCents ?? 0) / 100).toFixed(2));
   160→    } catch (e) {
   161→      toast({ title: "Could not load work order", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   162→      setDetailOpen(false);
   163→    } finally {
   164→      setDetailLoading(false);
   165→    }
   166→  }, [toast]);
   167→
   168→  const refreshDetail = useCallback(async (id: string) => {
   169→    try {
   170→      const res = await api.get<WODetail>(`/api/v1/work-orders/${id}`);
   171→      setDetail(res.data);
   172→    } catch { /* detail refresh is best-effort */ }
   173→  }, []);
   174→
   175→  const afterMutation = useCallback(async () => {
   176→    setReloadKey((k) => k + 1);
   177→    if (detail) await refreshDetail(detail.id);
   178→  }, [detail, refreshDetail]);
   179→
   180→  const runTransition = useCallback(async (action: string, extra?: Record<string, unknown>) => {
   181→    if (!detail) return;
   182→    setBusy(true);
   183→    try {
   184→      await api.post<WODetail>(`/api/v1/work-orders/${detail.id}/transition`, { action, ...extra });
   185→      toast({ title: "Success", description: `Work order ${detail.code} — ${humanize(action)} done.` });
   186→      if (action === "complete" || action === "cancel") {
   187→        setDetailOpen(false);
   188→        setDetail(null);
   189→      } else {
   190→        await refreshDetail(detail.id);
   191→      }
   192→      setReloadKey((k) => k + 1);
   193→    } catch (e) {
   194→      toast({ title: "Action failed", description: e instanceof Error ? e.message : "Please try again.", variant: "destructive" });
   195→    } finally {
   196→      setBusy(false);
   197→    }
   198→  }, [detail, refreshDetail, toast]);
   199→
   200→  // ── Detail derived state ──
   201→  const isAssignedTech = !!detail && !!user && detail.technician?.user?.id === user.id;
   202→  const canOperate = !!detail && (isAssignedTech || canUpdate);
   203→  const canExecute = !!detail && (isAssignedTech || canComplete);
   204→  const woStatus = detail?.status;
   205→  const woEditable = !!woStatus && !["COMPLETED", "CANCELLED"].includes(woStatus);
   206→  const canEditLabour = woStatus === "IN_PROGRESS" && canOperate;
   207→  const checklistDone = (detail?.checklist ?? []).filter((c) => c.done).length;
   208→
   209→  async function toggleChecklist(item: ChecklistItem, done: boolean) {
   210→    if (!detail) return;
   211→    setBusy(true);
   212→    try {
   213→      await api.patch(`/api/v1/work-orders/${detail.id}/checklist`, { itemId: item.id, done });
   214→      await refreshDetail(detail.id);
   215→    } catch (e) {
   216→      toast({ title: "Could not update checklist", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   217→    } finally {
   218→      setBusy(false);
   219→    }
   220→  }
   221→
   222→  async function addChecklistItem() {
   223→    if (!detail || !newItem.trim()) return;
   224→    setBusy(true);
   225→    try {
   226→      await api.post(`/api/v1/work-orders/${detail.id}/checklist`, { label: newItem.trim() });
   227→      setNewItem("");
   228→      await refreshDetail(detail.id);
   229→    } catch (e) {
   230→      toast({ title: "Could not add item", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   231→    } finally {
   232→      setBusy(false);
   233→    }
   234→  }
   235→
   236→  async function addMaterial() {
   237→    if (!detail) return;
   238→    if (!matForm.name.trim()) {
   239→      toast({ title: "Material name required", variant: "destructive" });
   240→      return;
   241→    }
   242→    const qty = Number(matForm.qty);
   243→    if (!isFinite(qty) || qty <= 0) {
   244→      toast({ title: "Quantity must be greater than zero", variant: "destructive" });
   245→      return;
   246→    }
   247→    setBusy(true);
   248→    try {
   249→      await api.post(`/api/v1/work-orders/${detail.id}/materials`, {
   250→        name: matForm.name.trim(),
   251→        quantity: qty,
   252→        unitCostCents: toCents(matForm.cost || "0"),
   253→      });
   254→      setMatForm({ name: "", qty: "1", cost: "0" });
   255→      await afterMutation();
   256→    } catch (e) {
   257→      toast({ title: "Could not add material", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   258→    } finally {
   259→      setBusy(false);
   260→    }
   261→  }
   262→
   263→  async function removeMaterial(materialId: string) {
   264→    if (!detail) return;
   265→    setBusy(true);
   266→    try {
   267→      await api.del(`/api/v1/work-orders/${detail.id}/materials${qs({ materialId })}`);
   268→      await afterMutation();
   269→    } catch (e) {
   270→      toast({ title: "Could not remove material", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   271→    } finally {
   272→      setBusy(false);
   273→    }
   274→  }
   275→
   276→  async function saveLabour() {
   277→    if (!detail) return;
   278→    const hours = Number(labourHours);
   279→    if (!isFinite(hours) || hours < 0) {
   280→      toast({ title: "Invalid hours", variant: "destructive" });
   281→      return;
   282→    }
   283→    setBusy(true);
   284→    try {
   285→      await api.patch(`/api/v1/work-orders/${detail.id}`, { labourHours: hours, labourRateCents: toCents(labourRate || "0") });
   286→      toast({ title: "Labour updated" });
   287→      await afterMutation();
   288→    } catch (e) {
   289→      toast({ title: "Could not save labour", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   290→    } finally {
   291→      setBusy(false);
   292→    }
   293→  }
   294→
   295→  // ── Create dialog ──
   296→  const [createOpen, setCreateOpen] = useState(false);
   297→  const [submitting, setSubmitting] = useState(false);
   298→  const [customers, setCustomers] = useState<CustomerOpt[]>([]);
   299→  const [equipment, setEquipment] = useState<EquipmentOpt[]>([]);
   300→  const [techs, setTechs] = useState<TechOpt[]>([]);
   301→  const [openComplaints, setOpenComplaints] = useState<ComplaintOpt[]>([]);
   302→  const draft = useDraft<WOCreateForm>({ formKey: "workorder.create", initial: EMPTY_CREATE });
   303→
   304→  useEffect(() => {
   305→    if (!createOpen || !canCreate) return;
   306→    let alive = true;
   307→    Promise.all([
   308→      api.get<CustomerOpt[]>(`/api/v1/customers${qs({ pageSize: 200 })}`).catch(() => null),
   309→      api.get<TechOpt[]>(`/api/v1/technicians${qs({ pageSize: 200 })}`).catch(() => null),
   310→      api.get<ComplaintOpt[]>(`/api/v1/complaints${qs({ status: "IN_PROGRESS", pageSize: 200 })}`).catch(() => null),
   311→    ]).then(([c, t, cmp]) => {
   312→      if (!alive) return;
   313→      setCustomers(c && Array.isArray(c.data) ? c.data : []);
   314→      setTechs(t && Array.isArray(t.data) ? t.data : []);
   315→      setOpenComplaints(cmp && Array.isArray(cmp.data) ? cmp.data : []);
   316→    });
   317→    return () => { alive = false; };
   318→  }, [createOpen, canCreate]);
   319→
   320→  useEffect(() => {
   321→    if (!createOpen || !draft.value.customerId) { setEquipment([]); return; }
   322→    let alive = true;
   323→    api.get<EquipmentOpt[]>(`/api/v1/equipment${qs({ customerId: draft.value.customerId, pageSize: 200 })}`)
   324→      .then((r) => { if (alive) setEquipment(Array.isArray(r.data) ? r.data : []); })
   325→      .catch(() => { if (alive) setEquipment([]); });
   326→    return () => { alive = false; };
   327→  }, [createOpen, draft.value.customerId]);
   328→
   329→  function updateMaterialRow(index: number, patch: Partial<WOMaterialForm>) {
   330→    draft.setValue({
   331→      materials: draft.value.materials.map((m, i) => (i === index ? { ...m, ...patch } : m)),
   332→    });
   333→  }
   334→
   335→  async function submitCreate() {
   336→    const v = draft.value;
   337→    if (v.title.trim().length < 3) {
   338→      toast({ title: "Check the form", description: "Title needs at least 3 characters.", variant: "destructive" });
   339→      return;
   340→    }
   341→    if (!v.customerId) {
   342→      toast({ title: "Check the form", description: "Select a customer.", variant: "destructive" });
   343→      return;
   344→    }
   345→    const materials = v.materials
   346→      .filter((m) => m.name.trim() && Number(m.qty) > 0)
   347→      .map((m) => ({ name: m.name.trim(), quantity: Number(m.qty), unitCostCents: toCents(m.cost || "0") }));
   348→    const checklist = v.checklistText.split("\n").map((s) => s.trim()).filter(Boolean);
   349→
   350→    setSubmitting(true);
   351→    try {
   352→      const res = await api.post<WORow>("/api/v1/work-orders", {
   353→        title: v.title.trim(),
   354→        description: v.description.trim() || undefined,
   355→        customerId: v.customerId,
   356→        equipmentId: v.equipmentId || undefined,
   357→        complaintId: v.complaintId || undefined,
   358→        technicianId: v.technicianId || undefined,
   359→        priority: v.priority,
   360→        scheduledDate: v.scheduledDate || undefined,
   361→        checklist,
   362→        materials,
   363→      });
   364→      toast({ title: "Work order created", description: `${res.data.code} created successfully.` });
   365→      draft.reset(EMPTY_CREATE);
   366→      setCreateOpen(false);
   367→      setReloadKey((k) => k + 1);
   368→    } catch (e) {
   369→      toast({ title: "Could not create work order", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   370→    } finally {
   371→      setSubmitting(false);
   372→    }
   373→  }
   374→
   375→  // ── Derived views ──
   376→  const stats = useMemo(() => {
   377→    const list = rows ?? [];
   378→    const count = (match: (r: WORow) => boolean) => list.filter(match).length;
   379→    return {
   380→      total: list.length,
   381→      pending: count((r) => PENDING_STATUSES.includes(r.status)),
   382→      inProgress: count((r) => r.status === "IN_PROGRESS"),
   383→      onHold: count((r) => r.status === "ON_HOLD"),
   384→      completed: count((r) => r.status === "COMPLETED"),
   385→      valueCents: list.filter((r) => !["CANCELLED"].includes(r.status)).reduce((s, r) => s + (r.totalCents ?? 0), 0),
   386→    };
   387→  }, [rows]);
   388→
   389→  const visibleRows = useMemo(() => {
   390→    const active = STATUS_TABS.find((t) => t.key === tab) ?? STATUS_TABS[0];
   391→    return (rows ?? []).filter((r) => active.match(r.status));
   392→  }, [rows, tab]);
   393→
   394→  const tabCount = (key: string) => {
   395→    const active = STATUS_TABS.find((t) => t.key === key);
   396→    if (!active) return 0;
   397→    return (rows ?? []).filter((r) => active.match(r.status)).length;
   398→  };
   399→
   400→  const columns: Column<WORow>[] = [
   401→    { key: "code", header: "Code", value: (r) => r.code, className: "font-mono text-xs whitespace-nowrap" },
   402→    { key: "title", header: "Title", value: (r) => r.title, className: "max-w-[240px] truncate" },
   403→    { key: "status", header: "Status", render: (r) => <StatusBadge status={r.status} />, value: (r) => r.status },
   404→    { key: "priority", header: "Priority", render: (r) => <PriorityBadge priority={r.priority} />, value: (r) => r.priority, hideOnMobile: true },
   405→    { key: "customer", header: "Customer", value: (r) => r.customer?.companyName ?? "", hideOnMobile: true },
   406→    { key: "technician", header: "Technician", value: (r) => r.technician?.user?.name ?? "", render: (r) => r.technician?.user?.name ?? "—" },
   407→    { key: "scheduledDate", header: "Scheduled", value: (r) => r.scheduledDate ?? "", render: (r) => fmtDate(r.scheduledDate), hideOnMobile: true },
   408→    { key: "totalCents", header: "Total", value: (r) => r.totalCents, render: (r) => money(r.totalCents), className: "tabular-nums whitespace-nowrap" },
   409→  ];
   410→
   411→  const complaintOptions = openComplaints.filter(
   412→    (c) => !draft.value.customerId || c.customer?.id === draft.value.customerId
   413→  );
   414→
   415→  return (
   416→    <div>
   417→      <PageHeader
   418→        title="Work Orders"
   419→        subtitle="Execution cockpit: assignments, checklists, materials and costs."
   420→        actions={canCreate ? (
   421→          <Button onClick={() => setCreateOpen(true)}>
   422→            <Plus className="h-4 w-4 mr-1.5" /> New Work Order
   423→          </Button>
   424→        ) : null}
   425→      />
   426→
   427→      {/* Stats */}
   428→      <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-6 gap-3 mb-5">
   429→        <StatCard title="Total" value={stats.total} icon={<ClipboardList className="h-5 w-5" />} loading={!rows && loading} />
   430→        <StatCard title="Pending" value={stats.pending} icon={<PlayCircle className="h-5 w-5" />} tone="warning" loading={!rows && loading} />
   431→        <StatCard title="In Progress" value={stats.inProgress} icon={<Wrench className="h-5 w-5" />} tone="warning" loading={!rows && loading} />
   432→        <StatCard title="On Hold" value={stats.onHold} icon={<PauseCircle className="h-5 w-5" />} tone="danger" loading={!rows && loading} />
   433→        <StatCard title="Completed" value={stats.completed} icon={<Hammer className="h-5 w-5" />} tone="success" loading={!rows && loading} />
   434→        <StatCard title="Total Value" value={money(stats.valueCents)} icon={<CircleDollarSign className="h-5 w-5" />} loading={!rows && loading} />
   435→      </div>
   436→
   437→      {/* Status tabs */}
   438→      <div className="flex flex-wrap gap-1.5 mb-4">
   439→        {STATUS_TABS.map((t) => (
   440→          <button
   441→            key={t.key}
   442→            onClick={() => setTab(t.key)}
   443→            className={cn(
   444→              "px-3 py-1.5 rounded-full text-xs font-medium border transition-colors",
   445→              tab === t.key ? "bg-primary text-primary-foreground border-primary" : "bg-background hover:bg-muted text-muted-foreground"
   446→            )}
   447→          >
   448→            {t.label} <span className="opacity-70 tabular-nums">({tabCount(t.key)})</span>
   449→          </button>
   450→        ))}
   451→      </div>
   452→
   453→      {loading && !rows ? (
   454→        <LoadingState label="Loading work orders…" rows={5} />
   455→      ) : loadError ? (
   456→        <ErrorState message={loadError} onRetry={load} />
   457→      ) : visibleRows.length === 0 ? (
   458→        <EmptyState
   459→          title="No work orders in this view"
   460→          hint={canCreate ? "Create a work order to dispatch a technician." : "Work orders will appear here when the team creates them."}
   461→          action={canCreate ? <Button variant="outline" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1.5" /> New Work Order</Button> : undefined}
   462→        />
