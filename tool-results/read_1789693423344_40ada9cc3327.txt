    28→    28→type CustomerRow = {
    29→    29→  id: string; code: string; companyName: string; contactPerson: string;
    30→    30→  email: string; phone: string; address: string; city: string; status: string; notes: string;
    31→    31→  createdAt: string;
    32→    32→  portalUser: { id: string; email: string; name: string; status: string } | null;
    33→    33→  _count: { equipment: number; complaints: number; invoices: number };
    34→    34→};
    35→    35→
    36→    36→type CustomerDetail = CustomerRow & {
    37→    37→  _count: { equipment: number; complaints: number; invoices: number; workOrders: number; quotations: number; payments: number };
    38→    38→  complaints: { id: string; code: string; title: string; status: string; priority: string; createdAt: string }[];
    39→    39→  invoices: { id: string; code: string; totalCents: number; paidCents: number; status: string; invoiceDate: string }[];
    40→    40→};
    41→    41→
    42→    42→// The DRAFT never stores secrets (see use-draft contract) — portalPassword is
    43→    43→// kept in transient state outside the persisted draft.
    44→    44→type FormState = {
    45→    45→  companyName: string; contactPerson: string; email: string; phone: string;
    46→    46→  address: string; city: string; notes: string; portalEmail: string;
    47→    47→};
    48→    48→
    49→    49→const EMPTY_FORM: FormState = {
    50→    50→  companyName: "", contactPerson: "", email: "", phone: "",
    51→    51→  address: "", city: "", notes: "", portalEmail: "",
    52→    52→};
    53→    53→
    54→    54→type FieldErrors = Record<string, string>;
    55→    55→
    56→    56→function extractFieldErrors(e: unknown): FieldErrors {
    57→    57→  if (e instanceof ClientApiError && Array.isArray(e.details)) {
    58→    58→    const out: FieldErrors = {};
    59→    59→    for (const d of e.details as { path?: string; message?: string }[]) {
    60→    60→      if (d?.path && d?.message) out[d.path] = d.message;
    61→    61→    }
    62→    62→    return out;
    63→    63→  }
    64→    64→  return {};
    65→    65→}
    66→    66→
    67→    67→function FieldError({ msg }: { msg?: string }) {
    68→    68→  if (!msg) return null;
    69→    69→  return <p className="text-xs text-destructive mt-1">{msg}</p>;
    70→    70→}

    72→    72→export function CustomersModule() {
    73→    73→  const { user } = useSession();

    76→    76→

    79→    79→  const [error, setError] = useState<string | null>(null);
    80→    80→
    81→    81→  // Create dialog + draft protection
    82→    82→  const [createOpen, setCreateOpen] = useState(false);

    84→    84→  const [fieldErrors, setFieldErrors] = useState<FieldErrors>({});

    86→    86→  const draft = useDraft<FormState>({ formKey: "customer.create", initial: EMPTY_FORM });
    87→    87→
    88→    88→  // Edit dialog

    90→    90→  const [editForm, setEditForm] = useState<FormState>(EMPTY_FORM);
    91→    91→
    92→    92→  // Detail dialog
    93→    93→  const [detailId, setDetailId] = useState<string | null>(null);

    95→    95→  const [detailLoading, setDetailLoading] = useState(false);
    96→    96→
    97→    97→  // Delete confirm
    98→    98→  const [deleteRow, setDeleteRow] = useState<CustomerRow | null>(null);

   102→   102→    setLoading(true);

   104→   104→    try {

   106→   106→      setRows(res.data);

   108→   108→      setError(e instanceof ClientApiError ? e.message : "Something went wrong. Please try again.");
   109→   109→    } finally {
   110→   110→      setLoading(false);

   112→   112→  }, []);
   113→   113→
   114→   114→  useEffect(() => { load(); }, [load]);
   115→   115→

   122→   122→  const openEdit = (row: CustomerRow) => {

   126→   126→      companyName: row.companyName, contactPerson: row.contactPerson, email: row.email,

   130→   130→  };
   131→   131→

   133→   133→    setDetailId(row.id);
   134→   134→    setDetail(null);
   135→   135→    setDetailLoading(true);
   136→   136→    api.get<CustomerDetail>(`/api/v1/customers/${row.id}`)
   137→   137→      .then((res) => setDetail(res.data))
   138→   138→      .catch((e) => toast({ title: "Could not load customer", description: e instanceof ClientApiError ? e.message : undefined, variant: "destructive" }))
   139→   139→      .finally(() => setDetailLoading(false));
   140→   140→  };

   142→   142→  async function submitCreate() {
   143→   143→    setSaving(true);

   147→   147→        companyName: draft.value.companyName,
   148→   148→        contactPerson: draft.value.contactPerson,
   149→   149→        email: draft.value.email,

   152→   152→        city: draft.value.city || undefined,
   153→   153→        notes: draft.value.notes || undefined,
   154→   154→        portalEmail: draft.value.portalEmail || undefined,

   158→   158→        title: "Customer created",
   159→   159→        description: `${res.data.companyName} (${res.data.code})${res.data.portalUser ? " — portal account ready" : ""}.`,

   162→   162→      setPortalPassword("");
   163→   163→      setCreateOpen(false);
   164→   164→      load();

   167→   167→      toast({ title: "Could not create customer", description: e instanceof ClientApiError ? e.message : undefined, variant: "destructive" });
   168→   168→    } finally {

   170→   170→    }

   172→   172→
   173→   173→  async function submitEdit() {

   175→   175→    setSaving(true);

   179→   179→        companyName: editForm.companyName,

   181→   181→        email: editForm.email,
   182→   182→        phone: editForm.phone,
   183→   183→        address: editForm.address || undefined,
   184→   184→        city: editForm.city || undefined,
   185→   185→        notes: editForm.notes || undefined,
   186→   186→      });
   187→   187→      toast({ title: "Customer updated", description: `${editForm.companyName} saved.` });
   188→   188→      setEditRow(null);
   189→   189→      load();
   190→   190→    } catch (e) {
   191→   191→      setFieldErrors(extractFieldErrors(e));
   192→   192→      toast({ title: "Could not update customer", description: e instanceof ClientApiError ? e.message : undefined, variant: "destructive" });
   193→   193→    } finally {
   194→   194→      setSaving(false);
   195→   195→    }
   196→   196→  }

   202→   202→      const res = await api.del<{ deleted?: boolean; softDeleted?: boolean }>(`/api/v1/customers/${deleteRow.id}`);
   203→   203→      toast({
   204→   204→        title: res.data.softDeleted ? "Customer deactivated" : "Customer deleted",
   205→   205→        description: res.data.softDeleted
   206→   206→          ? "The customer has linked records, so it was set to INACTIVE instead of removed."
   207→   207→          : `${deleteRow.companyName} removed.`,
   208→   208→      });
   209→   209→      setDeleteRow(null);
   210→   210→      load();
   211→   211→    } catch (e) {
   212→   212→      toast({ title: "Could not delete customer", description: e instanceof ClientApiError ? e.message : undefined, variant: "destructive" });
   213→   213→    } finally {
   214→   214→      setDeleting(false);
   215→   215→    }
   216→   216→  }
   217→   217→
   218→   218→  const totalEquipment = rows.reduce((s, r) => s + r._count.equipment, 0);
   219→   219→  const portalCount = rows.filter((r) => r.portalUser).length;
   220→   220→  const activeCount = rows.filter((r) => r.status === "ACTIVE").length;
   221→   221→
   222→   222→  const columns: Column<CustomerRow>[] = [
   223→   223→    { key: "code", header: "Code", value: (r) => r.code, className: "font-mono text-xs whitespace-nowrap" },
   224→   224→    {
   225→   225→      key: "companyName", header: "Company", value: (r) => r.companyName,
   226→   226→      render: (r) => (
   227→   227→        <div className="min-w-[160px]">
   228→   228→          <div className="font-medium truncate">{r.companyName}</div>
   229→   229→          <div className="text-xs text-muted-foreground truncate">{r.contactPerson}</div>
   230→   230→        </div>
   231→   231→      ),
   232→   232→    },
   233→   233→    {
   234→   234→      key: "email", header: "Contact", value: (r) => `${r.email} ${r.phone}`,
   235→   235→      render: (r) => (
   236→   236→        <div className="min-w-[150px]">
   237→   237→          <div className="text-sm truncate">{r.email}</div>
   238→   238→          <div className="text-xs text-muted-foreground">{r.phone}</div>
   239→   239→        </div>
   240→   240→      ),
   241→   241→      hideOnMobile: true,
   242→   242→    },
   243→   243→    { key: "city", header: "City", value: (r) => r.city, render: (r) => r.city || "—", hideOnMobile: true },
   244→   244→    { key: "equipment", header: "Equipment", value: (r) => r._count.equipment, render: (r) => r._count.equipment },
   245→   245→    { key: "complaints", header: "Complaints", value: (r) => r._count.complaints, render: (r) => r._count.complaints, hideOnMobile: true },
   246→   246→    { key: "invoices", header: "Invoices", value: (r) => r._count.invoices, render: (r) => r._count.invoices, hideOnMobile: true },
   247→   247→    {
   248→   248→      key: "portal", header: "Portal", value: (r) => (r.portalUser ? "yes" : "no"),
   249→   249→      render: (r) =>
   250→   250→        r.portalUser ? (
   251→   251→          <span className="inline-flex items-center gap-1 text-xs text-emerald-700 bg-emerald-100 rounded-full px-2 py-0.5">
   252→   252→            <Wifi className="h-3 w-3" /> {r.portalUser.email}
   253→   253→          </span>
   254→   254→        ) : (
   255→   255→          <span className="text-xs text-muted-foreground">—</span>
   256→   256→        ),
   257→   257→      hideOnMobile: true,
   258→   258→    },
   259→   259→    { key: "status", header: "Status", value: (r) => r.status, render: (r) => <StatusBadge status={r.status} /> },
   260→   260→    {
   261→   261→      key: "actions", header: "", sortable: false,
   262→   262→      render: (r) => (
   263→   263→        <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   264→   264→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openDetail(r)} aria-label={`View ${r.companyName}`}>

   266→   266→          </Button>
   267→   267→          {can("customers.update") ? (
   268→   268→            <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEdit(r)} aria-label={`Edit ${r.companyName}`}>
   269→   269→              <Pencil className="h-4 w-4" />
   270→   270→            </Button>
   271→   271→          ) : null}
   272→   272→          {can("customers.delete") ? (
   273→   273→            <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => setDeleteRow(r)} aria-label={`Delete ${r.companyName}`}>
   274→   274→              <Trash2 className="h-4 w-4" />
   275→   275→            </Button>
   276→   276→          ) : null}
   277→   277→        </div>
   278→   278→      ),
   279→   279→      className: "w-4",
   280→   280→    },
   281→   281→  ];
   282→   282→
   283→   283→  return (
   284→   284→    <div>
   285→   285→      <PageHeader
   286→   286→        title="Customers"
   287→   287→        subtitle="Corporate accounts, portal access and engagement overview"
   288→   288→        actions={
   289→   289→          can("customers.create") ? (
   290→   290→            <Button onClick={openCreate}>
   291→   291→              <Plus className="h-4 w-4 mr-1.5" /> New Customer
   292→   292→            </Button>
   293→   293→          ) : null
   294→   294→        }
   295→   295→      />
   296→   296→
   297→   297→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 mb-5">
   298→   298→        <StatCard title="Customers" value={rows.length} icon={<Building2 className="h-5 w-5" />} loading={loading} />
   299→   299→        <StatCard title="Active" value={activeCount} icon={<ClipboardList className="h-5 w-5" />} tone="success" loading={loading} />
   300→   300→        <StatCard title="Portal accounts" value={portalCount} icon={<KeyRound className="h-5 w-5" />} loading={loading} />
   301→   301→        <StatCard title="Equipment units" value={totalEquipment} icon={<Receipt className="h-5 w-5" />} loading={loading} />

   303→   303→
   304→   304→      {error ? (
   305→   305→        <ErrorState message={error} onRetry={load} />
   306→   306→      ) : loading ? (
   307→   307→        <LoadingState label="Loading customers…" />
   308→   308→      ) : (
   309→   309→        <DataTable
   310→   310→          columns={columns}
   311→   311→          rows={rows}
   312→   312→          rowKey={(r) => r.id}
   313→   313→          onRowClick={openDetail}
   314→   314→          searchPlaceholder="Search company, contact, email, code…"
   315→   315→          filters={[
   316→   316→            {
   317→   317→              key: "status", label: "Status",
   318→   318→              options: [{ value: "ACTIVE", label: "Active" }, { value: "INACTIVE", label: "Inactive" }],
   319→   319→              match: (r, v) => r.status === v,
   320→   320→            },
   321→   321→          ]}
   322→   322→          emptyTitle="No customers yet"
   323→   323→          emptyHint={can("customers.create") ? "Create your first customer to start tracking equipment and service." : "Customers will appear here once created."}
   324→   324→          exportName="customers"
   325→   325→        />
   326→   326→      )}
   327→   327→
   328→   328→      {/* Create dialog (draft-protected) */}
   329→   329→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   330→   330→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   331→   331→          <DialogHeader>
   332→   332→            <DialogTitle>New customer</DialogTitle>
   333→   333→            <DialogDescription>Code is generated automatically. Optionally provision a portal login.</DialogDescription>
   334→   334→          </DialogHeader>
   335→   335→
   336→   336→          {draft.draftExists ? (
   337→   337→            <div className="flex flex-wrap items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 px-3 py-2 text-sm">
   338→   338→              <span className="flex-1 min-w-[140px]">You have an unsent draft from a previous session.</span>
   339→   339→              <Button size="sm" variant="outline" onClick={draft.restore}>
   340→   340→                <RotateCcw className="h-3.5 w-3.5 mr-1" /> Restore draft
   341→   341→              </Button>
   342→   342→              <Button size="sm" variant="ghost" onClick={draft.discard}>Discard</Button>
   343→   343→            </div>
   344→   344→          ) : draft.dirty && draft.lastSavedAt ? (
   345→   345→            <p className="text-xs text-muted-foreground">Draft saved automatically at {draft.lastSavedAt.toLocaleTimeString()} — restored automatically if you leave.</p>
   346→   346→          ) : null}
   347→   347→
   348→   348→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   349→   349→            <div className="sm:col-span-2">
   350→   350→              <Label htmlFor="cus-name">Company name *</Label>

   352→   352→              <FieldError msg={fieldErrors.companyName} />
   353→   353→            </div>
   354→   354→            <div>
   355→   355→              <Label htmlFor="cus-contact">Contact person *</Label>
   356→   356→              <Input id="cus-contact" value={draft.value.contactPerson} onChange={(e) => draft.setValue({ contactPerson: e.target.value })} placeholder="Full name" />
   357→   357→              <FieldError msg={fieldErrors.contactPerson} />
   358→   358→            </div>
   359→   359→            <div>
   360→   360→              <Label htmlFor="cus-phone">Phone *</Label>
   361→   361→              <Input id="cus-phone" value={draft.value.phone} onChange={(e) => draft.setValue({ phone: e.target.value })} placeholder="+60 3-…" />
   362→   362→              <FieldError msg={fieldErrors.phone} />
   363→   363→            </div>
   364→   364→            <div>
   365→   365→              <Label htmlFor="cus-email">Email *</Label>
   366→   366→              <Input id="cus-email" type="email" value={draft.value.email} onChange={(e) => draft.setValue({ email: e.target.value })} placeholder="contact@company.my" />
   367→   367→              <FieldError msg={fieldErrors.email} />
   368→   368→            </div>
   369→   369→            <div>
   370→   370→              <Label htmlFor="cus-city">City</Label>
   371→   371→              <Input id="cus-city" value={draft.value.city} onChange={(e) => draft.setValue({ city: e.target.value })} placeholder="Kuala Lumpur" />
   372→   372→              <FieldError msg={fieldErrors.city} />
   373→   373→            </div>
   374→   374→            <div className="sm:col-span-2">
   375→   375→              <Label htmlFor="cus-address">Address</Label>
   376→   376→              <Input id="cus-address" value={draft.value.address} onChange={(e) => draft.setValue({ address: e.target.value })} placeholder="Street, building, postcode" />
   377→   377→              <FieldError msg={fieldErrors.address} />
   378→   378→            </div>
   379→   379→            <div className="sm:col-span-2">
   380→   380→              <Label htmlFor="cus-notes">Notes</Label>

   382→   382→              <FieldError msg={fieldErrors.notes} />
   383→   383→            </div>

   385→   385→            <div className="sm:col-span-2 rounded-lg border bg-muted/40 p-3 space-y-3">

   388→   388→              </p>
   389→   389→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
