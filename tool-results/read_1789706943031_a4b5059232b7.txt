   149→   149→    api.get<CustomerOpt[]>(`/api/v1/customers${qs({ customerId: cid })}`)
   150→   150→      .then((r) => {

   154→   154→          setSelectedCustomer(c);
   155→   155→          setCustQuery(c.companyName ?? c.name ?? c.id);

   158→   158→      .catch(() => undefined);
   159→   159→    return () => { alive = false; };
   160→   160→  }, [draft.value.customerId, isStaffUser, selectedCustomer?.id]);
   161→   161→
   162→   162→  // ── Equipment follows the selected customer (server-filtered) ──
   163→   163→  const [equipment, setEquipment] = useState<EquipmentOpt[]>([]);
   164→   164→  const [equipLoading, setEquipLoading] = useState(false);

   166→   166→
   167→   167→  useEffect(() => {

   171→   171→    setEquipLoading(true);

   204→   204→  }

   206→   206→  function saveDraft() {

   211→   211→  function pickCustomer(c: CustomerOpt) {

   225→   225→  function validate(): Record<string, string> {
   226→   226→    const errs: Record<string, string> = {};
   227→   227→    const t = draft.value.title.trim();
   228→   228→    if (!t) errs.title = "Title is required.";
   229→   229→    else if (t.length < 3) errs.title = "Title must be at least 3 characters.";
   230→   230→    const d = draft.value.description.trim();
   231→   231→    if (!d) errs.description = "Description is required.";
   232→   232→    else if (d.length < 3) errs.description = "Description must be at least 3 characters.";

   234→   234→    return errs;
   235→   235→  }

   237→   237→  async function createComplaint() {
   238→   238→    const errs = validate();
   239→   239→    setErrors(errs);
   240→   240→    if (Object.keys(errs).length > 0) {
   241→   241→      toast({ title: "Check the highlighted fields", variant: "destructive" });

   243→   243→    }
   244→   244→    setCreating(true);
   245→   245→    setSubmitError(null);

   247→   247→      const payload: Record<string, unknown> = {
   248→   248→        title: draft.value.title.trim(),
   249→   249→        description: draft.value.description.trim(),
   250→   250→        priority: draft.value.priority,
   251→   251→        ...(isStaffUser && draft.value.customerId ? { customerId: draft.value.customerId } : {}),
   252→   252→        ...(draft.value.equipmentId ? { equipmentId: draft.value.equipmentId } : {}),
   253→   253→      };
   254→   254→      const res = await api.post<{ id: string; code: string }>("/api/v1/complaints", payload);
   255→   255→      toast({ title: "Complaint created", description: `${res.data.code} logged successfully.` });
   256→   256→      draft.reset(EMPTY_CREATE);
   257→   257→      setSelectedCustomer(null);
   258→   258→      setCustQuery("");
   259→   259→      setPageDirty(false);
   260→   260→      // Continue the existing workflow on the complaint's dedicated detail page.
   261→   261→      navigateTo("complaints", [res.data.id]);
   262→   262→    } catch (e) {
   263→   263→      // CRITICAL: keep every user-entered value on failure — show the error and allow retry.
   264→   264→      const msg = e instanceof Error ? e.message : "Could not create the complaint. Please try again.";
   265→   265→      setSubmitError(msg);
   266→   266→      toast({ title: "Could not create complaint", description: msg, variant: "destructive" });
   267→   267→    } finally {
   268→   268→      setCreating(false);
   269→   269→    }
   270→   270→  }
   271→   271→
   272→   272→  if (!canCreate) {
   273→   273→    return (
   274→   274→      <EmptyState
   275→   275→        title="You don't have permission to create complaints"

   278→   278→    );
   279→   279→  }

   289→   289→      </Button>
   290→   290→    </>

   292→   292→
   293→   293→  return (

   300→   300→              onClick={goBackToList}
   301→   301→              className="hover:text-foreground hover:underline underline-offset-2 rounded"
   302→   302→            >
   303→   303→              Complaints
   304→   304→            </button>
   305→   305→          </li>
   306→   306→          <li aria-hidden><ChevronRight className="h-3.5 w-3.5" /></li>

   311→   311→      {/* Back link */}

   328→   328→          <span className="text-muted-foreground">
   329→   329→            Unsubmitted draft saved {draft.lastSavedAt ? fmtDateTime(draft.lastSavedAt) : "earlier"} — restore it to continue where you left off.

   332→   332→            <Button size="sm" variant="outline" onClick={draft.restore}>Restore</Button>
   333→   333→            <Button size="sm" variant="ghost" onClick={draft.discard}>Discard</Button>
   334→   334→          </div>
   335→   335→        </div>
   336→   336→      ) : null}
   337→   337→
   338→   338→      {submitError ? (
   339→   339→        <div role="alert" className="mb-4 flex items-start gap-2 rounded-lg border border-destructive/40 bg-destructive/5 p-3 text-sm text-destructive">
   340→   340→          <AlertCircle className="h-4 w-4 mt-0.5 shrink-0" aria-hidden />
   341→   341→          <div>
   342→   342→            <p className="font-medium">The complaint could not be created.</p>
   343→   343→            <p className="mt-0.5">{submitError} Your entries are preserved — you can retry.</p>
   344→   344→          </div>
   345→   345→        </div>
   346→   346→      ) : null}
   347→   347→
   348→   348→      <div className="grid gap-4 xl:grid-cols-5">
   349→   349→        {/* ── Complaint information ── */}
   350→   350→        <Card className="xl:col-span-3 shadow-sm">
   351→   351→          <CardHeader className="pb-3">
   352→   352→            <CardTitle className="text-base">Complaint Information</CardTitle>
   353→   353→          </CardHeader>
   354→   354→          <CardContent className="space-y-4">
   355→   355→            <div className="space-y-1.5">
   356→   356→              <div className="flex items-center justify-between">
   357→   357→                <Label htmlFor="nc-title">Title *</Label>
   358→   358→                <span className="text-[11px] text-muted-foreground tabular-nums">{draft.value.title.trim().length}/{TITLE_MAX}</span>

   369→   369→              {errors.title ? <p id="nc-title-err" className="text-xs text-destructive">{errors.title}</p> : null}
   370→   370→            </div>
   371→   371→
   372→   372→            <div className="space-y-1.5">
   373→   373→              <div className="flex items-center justify-between">
   374→   374→                <Label htmlFor="nc-desc">Description *</Label>
   375→   375→                <span className="text-[11px] text-muted-foreground tabular-nums">{draft.value.description.trim().length}/{DESC_MAX}</span>

   378→   378→                id="nc-desc"
   379→   379→                value={draft.value.description}

   382→   382→                rows={7}
   383→   383→                maxLength={DESC_MAX}
   384→   384→                aria-invalid={!!errors.description}
   385→   385→                aria-describedby={errors.description ? "nc-desc-err" : undefined}
   386→   386→                className="min-h-[9rem]"
   387→   387→              />
   388→   388→              {errors.description ? <p id="nc-desc-err" className="text-xs text-destructive">{errors.description}</p> : null}
   389→   389→            </div>
   390→   390→
   391→   391→            <div className="space-y-1.5 sm:max-w-xs">
   392→   392→              <Label>Priority</Label>

   394→   394→                <SelectTrigger aria-label="Priority"><SelectValue /></SelectTrigger>
   395→   395→                <SelectContent>
   396→   396→                  {PRIORITIES.map((p) => <SelectItem key={p} value={p}>{humanize(p)}</SelectItem>)}
   397→   397→                </SelectContent>
   398→   398→              </Select>
   399→   399→              <p className="text-[11px] text-muted-foreground">Defaults to Medium — the server enforces the same priority scale.</p>
   400→   400→            </div>
   401→   401→          </CardContent>
   402→   402→        </Card>
   403→   403→
   404→   404→        {/* ── Customer & asset ── */}
   405→   405→        <div className="xl:col-span-2 space-y-4">
   406→   406→          <Card className="shadow-sm">
   407→   407→            <CardHeader className="pb-3">
   408→   408→              <CardTitle className="text-base">Customer &amp; Asset</CardTitle>
   409→   409→            </CardHeader>
